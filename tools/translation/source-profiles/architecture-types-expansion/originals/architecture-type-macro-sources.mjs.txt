// Actual original declarative expansion DATA. This does not close architecture
// exports, resolve the macro owner, or prove absence of a competing type prefix.
import {types} from 'node:util';
import {originalFreshPrimitiveNamespaceSources,PRIMITIVE_NAMESPACE_SOURCE_PROFILE} from './extensions/primitive-namespace-sources.mjs';
import {createAttributedLocalDeclarativeMacroEngine} from './attributed-local-declarative-macro-engine.mjs';
import {originalMacroLiteralGrammar} from './macro-literal-source.mjs';
import {originalTopLevelMacroInvocations} from './macro-token-trees.mjs';
import {sourceItems} from './rust-structure.mjs';
import {transportConcreteSyntaxSource,transportConcreteSyntaxMeta} from './concrete-syntax-meta.mjs';
import {createRulePipeline} from './rule-pipeline.mjs';
const need=(value,message)=>{if(!value)throw new TypeError('Original architecture types source: '+message);};
export function proposeOriginalArchitectureTypeMacroSources(options){
 need(options&&typeof options==='object'&&!types.isProxy(options)&&Object.getPrototypeOf(options)===Object.prototype,'ordinary explicit source options required');const fields=Object.getOwnPropertyDescriptors(options);need(Reflect.ownKeys(fields).length===1&&Object.hasOwn(fields,'paths')&&Object.hasOwn(fields.paths,'value'),'only own complete selected source paths allowed');const paths=fields.paths.value;need(Array.isArray(paths)&&!types.isProxy(paths)&&Object.getPrototypeOf(paths)===Array.prototype&&paths.length>0&&paths.length<=32,'bounded dense original source paths required');const entries=Object.getOwnPropertyDescriptors(paths);need(Reflect.ownKeys(entries).length===paths.length+1,'dense own source path array required');const selected=[];for(let index=0;index<paths.length;index++){need(entries[index]&&Object.hasOwn(entries[index],'value'),'own original source path required');const path=entries[index].value;need(typeof path==='string'&&path.startsWith('crates/core_arch/src/')&&path.endsWith('.rs')&&!/[\\\0:]/u.test(path)&&path.split('/').every(part=>part&&part!=='.'&&part!=='..'),'actual original architecture source path required');selected.push(path);}need(new Set(selected).size===selected.length,'unique complete original source paths required');
 const originals=originalFreshPrimitiveNamespaceSources(),definitionPath='crates/core_arch/src/macros.rs',definitionInput=originals.get(definitionPath);need(definitionInput,'complete original types macro source required');const sourceInputs=new Map(),calls=[],names=[],remainingProducerObligations=[];let retained=0,roots=0,expansions=0,cells=0,engine,definition;
 const tick=()=>need(++cells<=200000,'complete original architecture source-cell bound');
 function source(path){tick();const pin=originals.get(path);need(pin&&typeof pin.source==='string'&&Buffer.byteLength(pin.source)<=8000000,'complete original architecture source input required');if(!sourceInputs.has(path)){retained+=Buffer.byteLength(pin.source);need(retained<=134217728,'complete original architecture input raw-source bound');const concrete=transportConcreteSyntaxSource(pin.source,{language:'Rust'});need(transportConcreteSyntaxMeta(concrete.meta).source===pin.source,'complete original architecture source CST inverse differs');sourceInputs.set(path,{...pin,concreteSyntax:concrete.meta,inverse:pin.source});}return pin.source;}
 function expand(proof,root,depth=0){
  tick();need(depth<=64&&++expansions<=128&&engine.isDeclarativeExpansion(proof),'actual bounded private original macro output required');const observed=engine.declarativeNamespaceObservation(proof),nested=engine.declarativeNestedMacroInvocations(proof),record={proof,lineage:engine.declarativeExpansionLineage(proof),observation:observed,children:[],remaining:[],ownerVerified:false,namespaceAuthority:false,runtimeAuthority:false};for(const declaration of observed.declarations){tick();names.push({...declaration,sourcePath:root.sourcePath,originalInvocationSpan:{...root.sourceSpan},expansionDigest:proof.digest,macroOwnerResolved:false,namespaceAuthority:false,sourceAuthority:false});}
  for(const invocation of nested){tick();if(invocation.path.length===1&&invocation.writtenPath.length===1&&invocation.writtenPath[0]==='types'&&invocation.hygiene==='written-or-forwarded-path'){const input=engine.declarativeNestedMacroInput(proof,invocation),child=engine.expandDeclarativeMacro(definition,input);record.children.push({invocation,expansion:expand(child,root,depth+1)});}else{const obligation={sourcePath:root.sourcePath,originalInvocationSpan:{...root.sourceSpan},invocation,parentExpansionDigest:proof.digest,ownerResolved:false,allOutputObligationsPending:true,namespaceAuthority:false};record.remaining.push(obligation);remainingProducerObligations.push(obligation);}}
  // Attributes/derives and hygienic macro ownership remain separate even when
  // the private expansion exposes actual root declaration names.
  remainingProducerObligations.push({sourcePath:root.sourcePath,originalInvocationSpan:{...root.sourceSpan},kind:'complete-attribute-derive-hygiene-and-export-closure',expansionDigest:proof.digest,ownerResolved:false,namespaceAuthority:false});return record;
 }
 const run=createRulePipeline([
  {phase:'language',rules:[
   {id:'rust:join-complete-original-types-definition-with-private-original-input-expansion',apply:()=>{
    const original=source(definitionPath);engine=createAttributedLocalDeclarativeMacroEngine(null,originalMacroLiteralGrammar());definition=engine.compileDeclarativeMacro(original,{name:'types',owner:['core','core_arch','macros']});const header=engine.declarativeDefinitionHeader(definition);need(header.kind==='macro_rules'&&header.name==='types','actual complete original types declaration required');for(const path of selected){const original=source(path),items=sourceItems(original);need(items.length<=20000,'complete original architecture item bound');remainingProducerObligations.push({sourcePath:path,kind:'complete-original-enclosing-owner-source-namespace-and-export-closure',ownerResolved:false,namespaceAuthority:false});for(const item of items){tick();const tokens=item.tokens.filter(row=>row.type!=='comment');if(!(tokens[0]?.text==='types!'||tokens[0]?.text==='types'&&tokens[1]?.text==='!'))continue;const fragmentOriginal=original.slice(item.start,item.end),invocations=originalTopLevelMacroInvocations(fragmentOriginal);need(invocations.length===1&&invocations[0].path.length===1&&invocations[0].path[0]==='types','one complete original plain types invocation required');const input=invocations[0],invocation={...input,fragmentOriginal,fragmentSourceSpan:{...input.sourceSpan},sourceFileSha256:originals.get(path).sha256,sourceSpan:{start:item.start+input.sourceSpan.start,end:item.start+input.sourceSpan.end}};need(++roots<=128,'complete original types root-invocation bound');const proof=engine.expandDeclarativeMacro(definition,invocation.trees);engine.declarativeExpansionOriginalInput(proof,invocation.trees);const root={sourcePath:path,...invocation};calls.push({sourcePath:path,originalInvocation:invocation,definitionHeader:header,expansion:expand(proof,root),sourceAuthority:false,semanticAuthority:false});}}return calls;
   }},
  ]},
  {phase:'repository',rules:[{id:'router:retain-full-original-architecture-CST-inputs-emitted-names-and-open-producers',apply:()=>{
   return {schema:'original-architecture-types-macro-source-proposal-v1',profileSha256:PRIMITIVE_NAMESPACE_SOURCE_PROFILE.manifestSha256,paths:selected,sourceInputs:[...sourceInputs.values()],definition,calls,emittedDeclarations:names,remainingProducerObligations,observedExpansions:expansions,observedSourceCells:cells,originalSourceUniverse:[...originals.values()].map(row=>({path:row.path,sha256:row.sha256,bytes:row.bytes})),macroOwnerResolved:false,typeNamespaceComplete:false,typePrefixAbsenceQualified:false,architectureExportsComplete:false,sourceAuthority:false,semanticAuthority:false,namespaceAuthority:false,runtimeAuthority:false,crossInvocationCache:false,receivingExecuted:false};
  }}]},
 ]).run(null);
 const current=originalFreshPrimitiveNamespaceSources();need(current.size===originals.size,'complete original namespace source universe changed');for(const [path,pin]of originals){const now=current.get(path);need(now&&now.source===pin.source&&now.sha256===pin.sha256,'complete original namespace source input changed');}return {...run.value,ruleTrace:run.trace};
}
