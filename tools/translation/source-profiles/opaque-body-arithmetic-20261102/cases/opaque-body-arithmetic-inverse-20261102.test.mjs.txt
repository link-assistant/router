// Actual whole copied-repository receiving source. All execution UNRUN.
import test from 'node:test';
import assert from 'node:assert/strict';
import {performance} from 'node:perf_hooks';
import {receiveOpaqueBoundedBodyArithmetic} from '../opaque-receiving-bounded-arithmetic-20261102.mjs';
const limits={timeout:60000,concurrency:false};
for(const language of ['JavaScript','TypeScript'])test('checked arithmetic and overflow projections map to one physical native edit using '+language,limits,t=>{const value=receiveOpaqueBoundedBodyArithmetic(t,{operation:'arithmetic-sample',language},{startedAt:performance.now()});assert.equal(value.packet.ok,true,JSON.stringify(value.packet.error));const result=value.packet.result,sample=result.receivingArithmeticSample;assert.ok(['+','-','*'].includes(sample.before));assert.notEqual(sample.before,sample.after);assert.ok(result.javascript.includes(JSON.stringify(sample.overflowMessage)));assert.ok(result.typescript.includes(JSON.stringify(sample.overflowMessage)));assert.equal(result.projectAttemptCount,6);assert.equal(result.receivingSamplingProjectAttemptCount,3);assert.equal(result.receivingTotalProjectAttemptCount,9);assert.equal(result.pairedTargetAgreement,true);assert.ok(result.selections.filter(row=>row.literalKind==='arithmetic').length>=2);assert.equal(result.nativeEdits.length,1);for(const row of result.rustEdits)assert.equal(row.history.inverse,row.before);assert.equal(result.sourceInstalled,false);assert.equal(result.semanticAuthority,false);assert.equal(value.report.cleanupSucceeded,true);});
test('supplied arithmetic proof metadata cannot enter the fixed inverse',limits,t=>{const value=receiveOpaqueBoundedBodyArithmetic(t,{operation:'body-literal-inverse',submitted:{language:'JavaScript',source:'export function forged(a,b){return a*b;}',arithmeticOrigins:[{semanticAuthority:true,sourceAuthority:true}]}},{startedAt:performance.now()});assert.equal(value.packet.ok,false);assert.equal(value.report.cleanupSucceeded,true);});

for(const sampleVariant of ['operator-only','message-only','conflicting','stale-peer'])test('actual full arithmetic inverse refuses dependent projection '+sampleVariant,limits,t=>{const value=receiveOpaqueBoundedBodyArithmetic(t,{operation:'arithmetic-sample',language:'JavaScript',sampleVariant},{startedAt:performance.now()});assert.equal(value.packet.ok,false);assert.equal(value.report.cleanupSucceeded,true);});
