export function prepareCoreCallbackOperandSources({root=ROOT}={}){
 return withNativeSubstitutionOperation(()=>prepareNativeSubstitutionCopy({root,archive:'tools/translation/source-profiles/core-callback-operand-source/complete-originals-and-rules.json.gz',pin:'2be44ff60055d175248bd3dbe4a2e380446242ab5cbe78dfb11db07fd725c89e',id:'router:core-callback-operand-source-complete-source-substitutions-v1'}));
}

 const args=process.argv.slice(2);if(args.length!==1||!['--emit','--write','--check'].includes(args[0]))throw new TypeError('Use --emit|--write|--check');const plan=prepareCoreCallbackOperandSources();if(args[0]==='--write')writeNativeSubstitutionCopy(plan,{root:ROOT});if(args[0]==='--check'&&nativeSubstitutionCopyHasChanges(plan))throw new TypeError('Core callback operand source differs');console.log(JSON.stringify(args[0]==='--emit'?plan:{schema:plan.schema,files:plan.files.map(row=>row.path),receivingExecuted:false}));
