export function planRemovePersistentClient(client/*:string*/,source/*:any*/,markerSource/*:any*/)/*:any*/{

   if(env.ANTHROPIC_BASE_URL===marker.anthropic_base_url)set(env,'ANTHROPIC_BASE_URL',text(marker.previous_anthropic_base_url));
