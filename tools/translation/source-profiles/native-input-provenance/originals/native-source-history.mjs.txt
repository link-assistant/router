// Reusable full native source equation DATA; never semantic/type/owner authority.
import {readFileSync} from 'node:fs';
import {resolve} from 'node:path';
import {fileURLToPath} from 'node:url';
import {gunzipSync} from 'node:zlib';
import {createHash} from 'node:crypto';
import {transportConcreteSyntaxSource,transportConcreteSyntaxMeta,applyConcreteSyntaxSubstitutions,serializeConcreteSyntaxMeta} from './concrete-syntax-meta.mjs';
import {transportConcreteSourceEquations} from './concrete-source-equations.mjs';
const ROOT=fileURLToPath(new URL('../../',import.meta.url));
const ARCHIVE='tools/translation/source-profiles/native-catalog-alignment/complete-native-history.json.gz',PIN='f57bc08c1a48f162e1394a95fd3f7b2850557ee77ffeedd81d5d102141fe54be';
const hash=value=>createHash('sha256').update(value).digest('hex'),need=(value,message)=>{if(!value)throw new TypeError('Native source history: '+message);};
function local(root,path){need(typeof path==='string'&&path.length<=4096&&!path.startsWith('/')&&!path.includes('\\')&&!path.split('/').some(word=>!word||word==='.'||word==='..'),'repository-relative file required');return resolve(root,path);}
function input(root,archive=ARCHIVE,pin=PIN){need(typeof pin==='string'&&/^[a-f0-9]{64}$/u.test(pin),'explicit archive pin required');const packed=readFileSync(local(root,archive));need(packed.length<=8388608&&hash(packed)===pin,'complete pinned history archive differs');const data=JSON.parse(gunzipSync(packed,{maxOutputLength:16777216}).toString('utf8'));need(data.schema==='native-complete-source-history-v1'&&Array.isArray(data.files)&&data.files.length>0&&data.files.length<=64,'bounded complete history rows required');return {packed,data};}
function equation(root,row){
 need(row.language==='Rust'&&typeof row.before==='string'&&typeof row.after==='string'&&hash(row.before)===row.beforeSha256&&hash(row.after)===row.afterSha256&&Buffer.byteLength(row.before)<=2097152&&Buffer.byteLength(row.after)<=2097152&&Array.isArray(row.languageEdits)&&row.languageEdits.length===0&&Array.isArray(row.repositoryEdits),'complete native source/rule inputs required');need(readFileSync(local(root,row.originalPath),'utf8')===row.before,'physical complete original source differs');
 const original=transportConcreteSyntaxSource(row.before,{language:row.language}),applied=applyConcreteSyntaxSubstitutions(original.document,{languageRules:[{id:'native-source:retain-complete-original-language',edits:row.languageEdits}],repositoryRules:[{id:row.ruleId,edits:row.repositoryEdits}]}),decoded=transportConcreteSyntaxMeta(serializeConcreteSyntaxMeta(applied.value));need(decoded.source===row.after,'complete ordered native source substitution differs');
 const stages=transportConcreteSourceEquations([{before:row.before,after:row.before,language:row.language},{before:row.before,after:row.after,language:row.language}]).map((stage,index)=>({...stage,phase:index?'repository':'language'}));need(stages[0].inverse===row.before&&stages[1].inverse===row.before,'complete native inverse differs');return {path:row.path,originalPath:row.originalPath,language:row.language,before:row.before,after:row.after,inverse:row.before,stages,trace:applied.trace,sourceAuthority:false,semanticAuthority:false,namespaceAuthority:false,rustExecuted:false};
}
function compose(rows,current){
 need(rows.length>0&&rows.length<=64,'bounded native equation chain required');const incoming=new Set(rows.map(row=>row.after)),heads=rows.filter(row=>!incoming.has(row.before));need(heads.length===1,'one complete native history head required');const chain=[],seen=new Set();let at=heads[0];while(at){need(!seen.has(at)&&at.before!==at.after,'acyclic changing native source equations required');seen.add(at);chain.push(at);const next=rows.filter(row=>row.before===at.after);need(next.length<=1,'native source history cannot branch');at=next[0];}need(chain.length===rows.length,'native source history has disconnected or duplicate equations');const versions=[chain[0].before,...chain.map(row=>row.after)];need(new Set(versions).size===versions.length,'native history source versions must be distinct');need(versions.includes(current),'unknown complete native current source');return {...chain[0],after:chain.at(-1).after,current,inverse:chain[0].before,stages:chain.flatMap(row=>row.stages),trace:chain.flatMap(row=>row.trace),historyChain:chain,versions};
}
export function prepareNativeSourceHistory({root=ROOT,archive=ARCHIVE,pin=PIN}={}){
 root=resolve(root);const {packed,data}=input(root,archive,pin),groups=new Map();for(const row of data.files){const value=equation(root,row),group=groups.get(row.path)??[];group.push(value);groups.set(row.path,group);}const files=[...groups].map(([path,rows])=>compose(rows,readFileSync(local(root,path),'utf8')));need(readFileSync(local(root,archive)).equals(packed),'history input changed');for(const row of files){need(readFileSync(local(root,row.path),'utf8')===row.current,'native source changed during construction');for(const step of row.historyChain)need(readFileSync(local(root,step.originalPath),'utf8')===step.before,'native original changed during construction');}return {schema:'native-complete-source-history-plan-v1',files,sourceAuthority:false,semanticAuthority:false,namespaceAuthority:false,sourceCache:false,rustExecuted:false,receivingExecuted:false};
}
export function currentNativeSourceProvenance(root,row){
 root=resolve(root);need(row&&typeof row.source==='string'&&typeof row.originalPath==='string','complete original provenance row required');need(readFileSync(local(root,row.originalPath),'utf8')===row.source,'complete original provenance source changed');const source=readFileSync(local(root,row.path),'utf8');if(source===row.source)return {path:row.path,originalPath:row.originalPath,originalSource:row.source,source,language:row.language??'Rust',nativeSourceHistory:null,sourceAuthority:false,semanticAuthority:false,namespaceAuthority:false};
 const plan=prepareNativeSourceHistory({root}),matches=plan.files.filter(value=>value.path===row.path&&value.versions.includes(row.source)&&value.current===source);need(matches.length===1,'current native source needs one explicit full original-to-current equation chain');const file=matches[0],start=file.versions.indexOf(row.source),end=file.versions.indexOf(source);need(end>start,'current native source must be a recorded successor');need((row.language??'Rust')===file.language,'current native source language differs');const joined=compose(file.historyChain.slice(start,end),source);return {path:row.path,originalPath:row.originalPath,originalSource:row.source,source,language:row.language??'Rust',nativeSourceHistory:joined,sourceAuthority:false,semanticAuthority:false,namespaceAuthority:false};
}
export function assertNativeSourceProvenanceCurrent(root,row){
 const fresh=currentNativeSourceProvenance(root,{path:row.path,originalPath:row.originalPath,source:row.originalSource,language:row.language});need(fresh.source===row.source&&JSON.stringify(fresh.nativeSourceHistory)===JSON.stringify(row.nativeSourceHistory),'complete native provenance changed during a caller operation');
}
