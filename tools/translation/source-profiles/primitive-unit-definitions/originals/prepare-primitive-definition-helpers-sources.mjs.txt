const ROOT=fileURLToPath(new URL('../../',import.meta.url));
export function preparePrimitiveDefinitionHelpersSources({root=ROOT}={}){
 return withNativeSubstitutionOperation(()=>prepareNativeSubstitutionCopy({root,archive:'tools/translation/source-profiles/primitive-definition-helpers/complete-originals-and-rules.json.gz',pin:'0fa46eb646a0c2fc264906008ff17029a23dc517807b879480e16f75b2b2ae9d',id:'router:primitive-definition-helpers-complete-source-substitutions-v1'}));
}

 const args=process.argv.slice(2);if(args.length!==1||!['--emit','--write','--check'].includes(args[0]))throw new TypeError('Use --emit|--write|--check');const plan=preparePrimitiveDefinitionHelpersSources();if(args[0]==='--write')writeNativeSubstitutionCopy(plan,{root:ROOT});if(args[0]==='--check'&&nativeSubstitutionCopyHasChanges(plan))throw new TypeError('Primitive definition helpers source differs');console.log(JSON.stringify(args[0]==='--emit'?plan:{schema:plan.schema,files:plan.files.map(row=>row.path),receivingExecuted:false}));
