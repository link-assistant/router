function lowerStandardFloatingExpression(row,parameters,arithmetic,calls,standard){
 const ts=ownershipCompiler(),source=row.source,file=row.file;let cells=0;
 const visit=(node,depth=0)=>{need(++cells<=100000&&depth<=160,'complete modified return expression bound');if(ts.isParenthesizedExpression(node))return '('+visit(node.expression,depth+1)+')';if(ts.isIdentifier(node)&&node.text==='NaN'&&!parameters.has(node.text))return 'f64::NAN';if(ts.isIdentifier(node)){need(parameters.has(node.text),'return expression references an unbound or nonparameter name');return parameters.get(node.text);}
  if(ts.isNumericLiteral(node))return floatingLiteralSource(node.getText(file)).rust;
  if(ts.isStringLiteral(node))return emitForwardLensRustString(node.text);if(node.kind===ts.SyntaxKind.TrueKeyword)return 'true';if(node.kind===ts.SyntaxKind.FalseKeyword)return 'false';if(ts.isBigIntLiteral(node))return BigInt(source.slice(node.getStart(file),node.end).slice(0,-1).replace(/_/g,'')).toString();
  if(ts.isCallExpression(node)&&ts.isIdentifier(node.expression)&&!parameters.has(node.expression.text)&&!node.typeArguments?.length&&!node.questionDotToken){
   const name=node.expression.text;need(!node.arguments.some(ts.isSpreadElement),'complete scalar call argument source required');if(standard.has(name))return standard.lower(name,node.arguments.map(argument=>visit(argument,depth+1)));if(!arithmetic.names.has(name))return calls.lower(name,node.arguments.map(argument=>visit(argument,depth+1)));if(name==='ml_fixed'){need(node.arguments.length===4&&!node.arguments.some(ts.isSpreadElement),'original checked range helper arity required');return visit(node.arguments[0],depth+1);}if(name==='ml_divide'){need(node.arguments.length===6&&!node.arguments.some(ts.isSpreadElement)&&ts.isStringLiteral(node.arguments[2])&&node.arguments[2].text==='trunc'&&[ts.SyntaxKind.TrueKeyword,ts.SyntaxKind.FalseKeyword].includes(node.arguments[4].kind),'actual truncating native division/remainder helper required');return '('+visit(node.arguments[0],depth+1)+(node.arguments[4].kind===ts.SyntaxKind.TrueKeyword?' % ':' / ')+visit(node.arguments[1],depth+1)+')';}need(false,'arithmetic helper requires its own original lowering');
  }
  if(ts.isConditionalExpression(node))return '(if '+visit(node.condition,depth+1)+' { '+visit(node.whenTrue,depth+1)+' } else { '+visit(node.whenFalse,depth+1)+' })';
  if(ts.isPrefixUnaryExpression(node)&&node.operator===ts.SyntaxKind.ExclamationToken)return '!('+visit(node.operand,depth+1)+')';if(ts.isPrefixUnaryExpression(node)&&node.operator===ts.SyntaxKind.MinusToken&&ts.isBigIntLiteral(node.operand))return (-BigInt(source.slice(node.operand.getStart(file),node.operand.end).slice(0,-1).replace(/_/g,''))).toString();
  if(ts.isPrefixUnaryExpression(node)&&node.operator===ts.SyntaxKind.MinusToken)return '-('+visit(node.operand,depth+1)+')';
  if(ts.isBinaryExpression(node)){const operator=node.operatorToken.getText(file),selected={'===':'==','!==':'!=','<':'<','<=':'<=','>':'>','>=':'>=','&&':'&&','||':'||','+':'+','-':'-','*':'*','/':'/','%':'%'}[operator];need(selected,'return expression operator needs its own checked native lowering');return '('+visit(node.left,depth+1)+' '+selected+' '+visit(node.right,depth+1)+')';}
  need(false,'return expression requires actual scalar literals parameters predicates or Boolean negation');
 };return visit(row.expression);
}

function lowerStandardFloatingBody(change,parameters,arithmetic,calls,proof,iterator,standard){
 const row=change.after,ts=ownershipCompiler(),before=change.before,original=[...before.expression.statements],requested=[...row.expression.statements],hints=originalFloatingLocalTypes(proof);let guards=0;standard.assertFunctionScope(row);
 for(const statement of original){if(!ts.isExpressionStatement(statement)||!ts.isCallExpression(statement.expression)||!ts.isIdentifier(statement.expression.expression)||statement.expression.expression.text!=='ml_fixed'||!arithmetic.names.has('ml_fixed')||statement.expression.arguments.length!==4||!ts.isIdentifier(statement.expression.arguments[0])||!parameters.has(statement.expression.arguments[0].text))break;need(requested[guards]&&statement.getText(before.file)===requested[guards].getText(row.file),'original native parameter guards must remain exact');guards++;}
 const expression=(node,scope)=>lowerStandardFloatingExpression({...row,expression:node},scope,arithmetic,calls,standard);let cells=0;
 const block=(statements,outer,outerMutable,loopDepth,depth=0)=>{
  need(++cells<=100000&&depth<=128,'bounded complete scalar imperative block');const scope=new Map(outer),mutable=new Set(outerMutable),parts=[];let terminated=false;
  for(const statement of statements){need(++cells<=100000&&!terminated,'unreachable scalar source needs a separate preservation rule');
   if(ts.isVariableStatement(statement)){
    const flags=statement.declarationList.flags&ts.NodeFlags.BlockScoped;need(!statement.modifiers?.length&&[ts.NodeFlags.Const,ts.NodeFlags.Let].includes(flags)&&statement.declarationList.declarations.length===1,'one initialized block-scoped scalar declaration');const node=statement.declarationList.declarations[0];need(ts.isIdentifier(node.name)&&node.initializer&&!node.exclamationToken&&/^[A-Za-z_][A-Za-z0-9_]*$/u.test(node.name.text)&&!CONTROL_RESERVED.has(node.name.text)&&!scope.has(node.name.text),'fresh ordinary scalar binding without shadowing');const value=expression(node.initializer,scope),isMutable=flags===ts.NodeFlags.Let,hint=hints.get(node.name.text);parts.push('let '+(isMutable?'mut ':'')+node.name.text+(hint?': '+hint:'')+' = '+value+';');scope.set(node.name.text,node.name.text);if(isMutable)mutable.add(node.name.text);continue;
   }
   if(ts.isExpressionStatement(statement)){
    const node=statement.expression;need(ts.isBinaryExpression(node)&&node.operatorToken.kind===ts.SyntaxKind.EqualsToken&&ts.isIdentifier(node.left)&&scope.has(node.left.text)&&mutable.has(node.left.text),'assignment needs its actual mutable scalar local');parts.push(scope.get(node.left.text)+' = '+expression(node.right,scope)+';');continue;
   }
   if(ts.isReturnStatement(statement)){need(statement.expression,'complete scalar function return value');parts.push('return '+expression(statement.expression,scope)+';');terminated=true;continue;}
   if(ts.isBreakStatement(statement)||ts.isContinueStatement(statement)){need(loopDepth>0&&!statement.label,'unlabelled scalar control requires its actual loop');parts.push(ts.isBreakStatement(statement)?'break;':'continue;');terminated=true;continue;}
   if(ts.isIfStatement(statement)){
    need(ts.isBlock(statement.thenStatement)&&(!statement.elseStatement||ts.isBlock(statement.elseStatement)),'native scalar conditional requires complete blocks');const yes=block([...statement.thenStatement.statements],scope,mutable,loopDepth,depth+1),no=statement.elseStatement?block([...statement.elseStatement.statements],scope,mutable,loopDepth,depth+1):{source:'',terminated:false};parts.push('if '+expression(statement.expression,scope)+' {\n'+yes.source+'\n} else {\n'+no.source+'\n}');terminated=yes.terminated&&no.terminated;continue;
   }
   if(ts.isForOfStatement(statement)){
    need(!statement.awaitModifier&&ts.isBlock(statement.statement)&&ts.isVariableDeclarationList(statement.initializer)&&statement.initializer.declarations.length===1,'ordinary complete synchronous range for-of');const flags=statement.initializer.flags&ts.NodeFlags.BlockScoped,binding=statement.initializer.declarations[0];need([ts.NodeFlags.Const,ts.NodeFlags.Let].includes(flags)&&ts.isIdentifier(binding.name)&&!binding.initializer&&!binding.type&&!binding.exclamationToken&&/^[A-Za-z_][A-Za-z0-9_]*$/u.test(binding.name.text)&&!CONTROL_RESERVED.has(binding.name.text)&&!scope.has(binding.name.text),'fresh original scalar range binding');
    const call=statement.expression;need(ts.isCallExpression(call)&&ts.isIdentifier(call.expression)&&!scope.has(call.expression.text)&&iterator.names.has(call.expression.text)&&!call.questionDotToken&&!call.typeArguments?.length&&!call.arguments.some(ts.isSpreadElement),'actual original registered range helper call');const reversed=call.expression.text==='ml_router_reverse_range';need(call.arguments.length===(reversed?4:3)&&[ts.SyntaxKind.TrueKeyword,ts.SyntaxKind.FalseKeyword].includes(call.arguments[2].kind),'complete actual range/inclusive arguments');if(reversed)need(ts.isArrowFunction(call.arguments[3])&&call.arguments[3].parameters.length===1&&ts.isIdentifier(call.arguments[3].parameters[0].name)&&!ts.isBlock(call.arguments[3].body),'original range retreat callback syntax; complete reproduction remains required');
    const hint=hints.get(binding.name.text);need(typeof hint==='string'&&/^(?:[ui](?:8|16|32|64|128)|[ui]size)$/u.test(hint),'actual original fixed range binding type observation required');
    const endpoint=node=>{let raw=node;while(ts.isParenthesizedExpression(raw))raw=raw.expression;let operand=ts.isPrefixUnaryExpression(raw)?raw.operand:null;while(operand&&ts.isParenthesizedExpression(operand))operand=operand.expression;const negative=ts.isPrefixUnaryExpression(raw)&&raw.operator===ts.SyntaxKind.MinusToken&&ts.isBigIntLiteral(operand),literal=negative?operand:raw;if(ts.isBigIntLiteral(literal)){const value=BigInt(literal.getText(row.file).slice(0,-1).replace(/_/gu,''))*(negative?-1n:1n);return value.toString()+'_'+hint;}return expression(node,scope);};
    const start=endpoint(call.arguments[0]),end=endpoint(call.arguments[1]),rangeSource=start+(call.arguments[2].kind===ts.SyntaxKind.TrueKeyword?'..=':'..')+end,range=reversed?'('+rangeSource+').rev()':rangeSource,innerScope=new Map(scope).set(binding.name.text,binding.name.text),innerMutable=new Set(mutable);if(flags===ts.NodeFlags.Let)innerMutable.add(binding.name.text);const inner=block([...statement.statement.statements],innerScope,innerMutable,loopDepth+1,depth+1);parts.push('for '+(flags===ts.NodeFlags.Let?'mut ':'')+binding.name.text+' in '+range+' {\n'+inner.source+'\n}');continue;
   }
   if(ts.isWhileStatement(statement)){
    need(ts.isBlock(statement.statement),'native scalar while requires a complete block');const inner=block([...statement.statement.statements],scope,mutable,loopDepth+1,depth+1);parts.push((statement.expression.kind===ts.SyntaxKind.TrueKeyword?'loop':'while '+expression(statement.expression,scope))+' {\n'+inner.source+'\n}');continue;
   }
   need(false,'scalar imperative statement needs its own checked rule');
  }
  return {source:parts.join('\n'),terminated};
 };
 const result=block(requested.slice(guards),parameters,new Set(),0);need(result.terminated,'complete scalar function paths must return values');return result.source;
}

export function proposeForwardStandardFloatingBodySynchronization(options){
 assertReverseConcreteSourceHost();need(options&&typeof options==='object'&&!types.isProxy(options),'ordinary explicit synchronization input required');const allowed=new Set(['entry','paths','readSource','javascript','typescript','compilationProfile','jsonMacroDependencies','deriveDependencies','cargoManifest']);const fields=Object.getOwnPropertyDescriptors(options);for(const key of Reflect.ownKeys(fields))need(typeof key==='string'&&allowed.has(key)&&Object.hasOwn(fields[key],'value'),'unknown or accessor synchronization option');const input=Object.fromEntries(Object.entries(fields).map(([key,field])=>[key,field.value]));
 const entry=path(input.entry);need(Array.isArray(input.paths)&&input.paths.length>0&&input.paths.length<=4096&&!types.isProxy(input.paths)&&Object.getPrototypeOf(input.paths)===Array.prototype,'complete bounded Rust path universe required');const pathFields=Object.getOwnPropertyDescriptors(input.paths);need(Reflect.ownKeys(pathFields).length===input.paths.length+1,'dense ordinary source path list required');const paths=[];for(let index=0;index<input.paths.length;index++){const field=pathFields[index];need(field&&Object.hasOwn(field,'value'),'source paths require own data entries');paths.push(path(field.value));}need(typeof input.readSource==='function'&&!types.isProxy(input.readSource),'actual ordinary source reader required');need(typeof input.javascript==='string'&&input.typescript===undefined||typeof input.typescript==='string'&&input.javascript===undefined,'exactly one complete proposed JavaScript or TypeScript source required');const language=input.typescript===undefined?'JavaScript':'TypeScript',targetKey=language==='TypeScript'?'typescript':'javascript',requestedSource=input[targetKey];need(new Set(paths).size===paths.length&&paths.includes(entry),'unique complete source paths including entry required');
 const sources=new Map();let bytes=0;for(const name of paths){const source=input.readSource(name);need(typeof source==='string'&&source.length<=8388608&&!/\p{Surrogate}/u.test(source),'bounded actual UTF8 Rust source required');bytes+=Buffer.byteLength(source);need(bytes<=134217728,'whole receiving source byte budget');sources.set(name,source);}
 const current=()=>{assertReverseConcreteSourceHost();for(const [name,source]of sources)need(input.readSource(name)===source,'original receiving Rust source changed during the proposal');};
 const compile=values=>translateProjectWithFlatSemanticMeta({entry,paths,readSource:name=>{need(values.has(name),'compiler requested an uncaptured original source');return values.get(name);},compilationProfile:input.compilationProfile,jsonMacroDependencies:input.jsonMacroDependencies,deriveDependencies:input.deriveDependencies,cargoManifest:input.cargoManifest});
 let original,originalCensus,originalTargets,bindings,originalTarget,requestedTarget,changes,edits,afterSources,proposed,arithmetic,calls,iterator,standard;
 const result=createRulePipeline([
  {phase:'language',rules:[
   {id:'standard-floating-body:recompile-original-project-and-bind-actual-scalar-target-parameters',apply:()=>{current();original=compile(sources);originalCensus=census(original);originalTargets={javascript:original.javascript,typescript:original.typescript};bindings=new Map();const functions=new Set(original.meta.executable.filter(row=>row.term==='fn').map(row=>JSON.stringify([row.source,row.name])));for(const declaration of original.executableDeclarations){if(!functions.has(JSON.stringify([declaration.source,declaration.qualifiedName])))continue;const binding=declaration.targetBinding('floating-scalars');if(binding.schema!=='original-checked-project-target-binding-v1'||binding.signatureFormat!=='floating-scalars-v1'||binding.source!==declaration.source||binding.qualifiedName!==declaration.qualifiedName||!binding.signature||!floatingScalarType(binding.signature.ret)||binding.signature.params.some(param=>!floatingScalarType(param.type)))continue;need(!bindings.has(binding.javascriptName),'ambiguous original scalar target function');bindings.set(binding.javascriptName,{...binding,declaration});}calls=createReturnCallSource(bindings,sources);originalTarget=parseControlBodyTarget(original[targetKey],language,bindings);arithmetic=captureReturnArithmeticSource(original.javascript,ownershipCompiler(),language==='JavaScript'?originalTarget.file:null);standard=createStandardFloatingCallSource({entry,source:original[targetKey],file:originalTarget.file});iterator=captureReturnIteratorSource(original.javascript,ownershipCompiler(),language==='JavaScript'?originalTarget.file:null);requestedTarget=parseControlBodyTarget(requestedSource,language,bindings);changes=completeControlBodyChanges(originalTarget,requestedTarget);return changes;}},
   {id:'standard-floating-body:lower-complete-native-target-formulas-into-original-source-bodies',apply:()=>{
    edits=[];let semanticChanges=0;const sourceNames=new Set();for(const change of changes){const binding=bindings.get(change.before.name),proof=binding.declaration.checkedSource();need(proof.schema==='original-checked-executable-source-v1'&&proof.source===binding.source&&proof.qualifiedName===binding.qualifiedName,'fresh actual checked function source required');const signature=proof.declaration;need(signature.params.length===change.after.parameters.length&&signature.params.every(param=>typeof param.name==='string'&&/^[A-Za-z_][A-Za-z0-9_]*$/u.test(param.name)&&param.name!=='_'),'actual ordinary Rust source parameter names required');const parameters=new Map(change.after.parameters.map((name,index)=>[name,signature.params[index].name]));need(parameters.size===signature.params.length,'unique native target parameter bindings required');const source=sources.get(proof.source),span=originalControlBodyInterval(source,proof);need([...parameters.values()].every(name=>!CONTROL_RESERVED.has(name)),'ordinary original Rust parameter spelling');const expression=lowerStandardFloatingBody(change,parameters,arithmetic,calls,proof,iterator,standard),key=proof.source+':'+span.join(':');need(!sourceNames.has(key),'one source body cannot receive competing target edits');sourceNames.add(key);const replacement='\n'+expression+'\n';edits.push({path:proof.source,name:proof.qualifiedName,kind:'standard-floating-body',span,before:source.slice(...span),replacement});if(JSON.stringify(floatingBodySignature(change.before.expression,ownershipCompiler(),change.before.source,change.before.file))!==JSON.stringify(floatingBodySignature(change.after.expression,ownershipCompiler(),change.after.source,change.after.file)))semanticChanges++;}need(semanticChanges>0,'a semantic scalar body expression edit is required');

   }},
   {id:'standard-floating-body:recompile-entire-project-and-reproduce-complete-target-with-native-formula-grammar',apply:()=>{for(const binding of bindings.values())binding.declaration=null;original=null;originalTarget=null;changes=null;const changedFunctionNames=new Set(edits.map(edit=>edit.name));proposed=compile(afterSources);need(JSON.stringify(originalCensus)===JSON.stringify(census(proposed)),'complete return project executable/refusal census changed');const expected=requestedTarget,actual=parseControlBodyTarget(proposed[targetKey],language,bindings);need(expected.rows.length===actual.rows.length,'whole modified target return membership differs');let oldEnd=0,newEnd=0;for(let index=0;index<expected.rows.length;index++){const left=expected.rows[index],right=actual.rows[index];need(left.name===right.name&&expected.source.slice(oldEnd,left.start)===actual.source.slice(newEnd,right.start),'whole modified target outside return expressions differs');const changed=changedFunctionNames.has(bindings.get(left.name).qualifiedName);if(!changed)need(left.source.slice(left.start,left.end)===right.source.slice(right.start,right.end),'unchanged complete native function body differs');else need(JSON.stringify(floatingBodySignature(left.expression,expected.ts,left.source,left.file))===JSON.stringify(floatingBodySignature(right.expression,actual.ts,right.source,right.file)),'whole modified native control body differs');oldEnd=left.end;newEnd=right.end;}need(expected.source.slice(oldEnd)===actual.source.slice(newEnd),'complete modified target suffix differs');return proposed;}},
  ]},
  {phase:'repository',rules:[{id:'router:retain-whole-return-source-equations-and-canonical-paired-targets',apply:()=>{
   current();arithmetic.verify();iterator.verify();const rustEdits=[];for(const [path,after]of afterSources){const before=sources.get(path);if(before===after)continue;const history=transportConcreteSourceEquations([{before,after,language:'Rust'}])[0];need(history.inverse===before,'whole return native inverse differs');rustEdits.push({path,before,after,history,expressionEdits:edits.filter(row=>row.path===path)});}const beforeJavaScript=language==='JavaScript'?requestedSource:originalTargets.javascript,beforeTypeScript=language==='TypeScript'?requestedSource:originalTargets.typescript,javascriptHistory=transportConcreteSourceEquations([{before:beforeJavaScript,after:proposed.javascript,language:'JavaScript'}])[0],typescriptHistory=transportConcreteSourceEquations([{before:beforeTypeScript,after:proposed.typescript,language:'TypeScript'}])[0];need(javascriptHistory.inverse===beforeJavaScript&&typescriptHistory.inverse===beforeTypeScript,'whole return paired inverse differs');return {schema:'forward-qualified-standard-floating-body-sync-proposal-v1',entry,sourceInputs:[...sources].map(([path,source])=>({path,source})),rustEdits,javascript:proposed.javascript,typescript:proposed.typescript,requestedSource:{language,source:requestedSource},javascriptHistory,typescriptHistory,expressionEditCount:edits.length,arithmeticSourceInputs:arithmetic.evidence,iteratorSourceInputs:iterator.evidence,calleeSourceInputs:calls.evidence(),standardRuntimeSourceInputs:standard.evidence(),ruleScope:'safe-nongeneric-source-f64-bodies-and-original-bound-standard-inherent-calls',executable:census(proposed).executable,refused:census(proposed).refused,sourceAuthority:false,semanticAuthority:false,runtimeParity:false,nativeRustGrammarChecked:false,receivingExecuted:false,crossInvocationCache:false};

}
