function recloseFull(before){let failed=false,primary;const check=run=>{try{run();}catch(error){if(!failed){failed=true;primary=error;}}};for(const row of before.files)check(()=>{if(!readCompactSparsePhysicalFile(row).equals(row.raw))throw TypeError('full physical compiler-bank bytes changed');});for(const row of before.directories)check(()=>{if(!sameCompactPhysicalSourceStamp(row.stat,lstatSync(row.path,{bigint:true})))throw TypeError('full physical compiler-bank ancestor changed');});check(()=>{const now=compactSparsePhysicalCensus(before.root,{maximum:402653184});if(JSON.stringify(now.files.map(row=>row.name))!==JSON.stringify(before.files.map(row=>row.name))||JSON.stringify(now.directories.map(row=>row.name))!==JSON.stringify(before.directories.map(row=>row.name)))throw TypeError('full physical compiler-bank namespace changed');});if(failed)throw primary;}
const fail=message=>{throw TypeError('Complete sparse Git source bank DATA: '+message);};
const oid=(raw,kind)=>createHash(kind).update(Buffer.from('blob '+raw.length+'\0')).update(raw).digest('hex');
function selection(input,available){if(types.isProxy(input)||!Array.isArray(input)||Object.getPrototypeOf(input)!==Array.prototype||input.length>8192||Reflect.ownKeys(input).length!==input.length+1)fail('bounded dense explicit hot source selection required');const rows=[],seen=new Set();for(let at=0;at<input.length;at++){const field=Object.getOwnPropertyDescriptor(input,String(at));if(!field||!Object.hasOwn(field,'value')||typeof field.value!=='string'||seen.has(field.value)||!available.has(field.value))fail('unique actual hot source paths required');seen.add(field.value);rows.push(field.value);}return rows.sort();}
function verifyBatchOutput(output,rows){let at=0;for(const row of rows){const end=output.indexOf(10,at);if(end<at||end-at>256)fail('complete actual Git batch header absent');const expected=row.blob+' blob '+row.raw.length;if(output.subarray(at,end).toString('ascii')!==expected)fail('actual Git source object header differs');const start=end+1,stop=start+row.raw.length;if(stop>=output.length||output[stop]!==10||!output.subarray(start,stop).equals(row.raw))fail('complete actual stored Git object bytes differ');at=stop+1;}if(at!==output.length)fail('unexpected actual Git batch suffix');}
/** All source blobs are real objects/paths in one fresh committed source-DATA
 * repository. Cold paths get genuine skip-worktree bits; supported source
 * readers may reissue their original sparse Git fallback. APIs that require
 * physical files still need explicitly materialized hot inputs before epochs.
 * Missing-resource recovery must restart the child, never restore live facts.
 */
export function createReadonlyProfileCompilerSourceBank({vaultDirectory,output,workspaceRoot,hotPaths=[],objectFormat='sha1'}={}){
 return withSourceInputValidation(()=>{
 if(types.isProxy(hotPaths)||!Array.isArray(hotPaths)||hotPaths.length)fail('complete actual non-profile hot closure is fixed by this new storage profile');if(typeof vaultDirectory!=='string'||typeof output!=='string'||!output||!['sha1','sha256'].includes(objectFormat))fail('explicit physical vault/fresh source-bank paths and object format required');const sourceStat=lstatSync(vaultDirectory,{bigint:true});if(!sourceStat.isDirectory()||sourceStat.isSymbolicLink())fail('ordinary physical input vault required');const inputRoot=realpathSync(vaultDirectory),requested=resolve(output),root=join(realpathSync(dirname(requested)),basename(requested));if(root===inputRoot||root.startsWith(inputRoot+sep)||inputRoot.startsWith(root+sep))fail('input vault and fresh source bank must be separate');try{lstatSync(root);fail('fresh source-bank output already exists');}catch(error){if(ownSourceFailureField(error,'code')!=='ENOENT')throw error;}

 return withCompleteSourceClosingFence(()=>withProtectedNativeWriter(inputRoot,()=>mutateProtectedNativeAbsoluteEntries(inputRoot,[root],captured=>{if(captured.length!==1||captured[0].canonicalTarget!==root)fail('actual admitted fresh compiler bank differs');

 const close=()=>{let failed=false,primary;const check=run=>{try{run();}catch(error){if(!failed){failed=true;primary=error;}}};if(inputWatch)check(()=>recloseCompactSparsePhysicalCensus(inputWatch));if(outputWatch)check(()=>recloseFull(outputWatch));if(commit)check(()=>{if(run(['rev-parse','--verify','HEAD']).trim()!==commit)fail('actual compiler-bank commit changed');});if(failed)throw primary;};
 try{return withCompleteSourceClosingFence(()=>{
 inputWatch=compactSparsePhysicalCensus(inputRoot,{retainBytes:true});model=readCompactSourceVaultDirectory(inputRoot);const inspected=inspectCompactSourceVault(model),available=new Map(inspected.metadata.files.map(row=>[row.path,row])),hot=selection([...available.keys()].filter(path=>!path.startsWith('tools/translation/source-profiles/')||path==='tools/translation/source-profiles/exact-source-successor-adoption-20261103/index.json'),available),hotSet=new Set(hot);if(!available.size)fail('complete nonempty original source bank required');for(const name of available.keys())if(name.split('/').includes('.git'))fail('vault cannot install Git control paths');const descriptions=inspected.metadata.files.slice().sort((a,b)=>a.path<b.path?-1:a.path>b.path?1:0),indexReserve=1048576+descriptions.reduce((sum,row)=>sum+256+Buffer.byteLength(row.path)*2,0),budget=402653184;
 const occupied=()=>compactSparsePhysicalCensus(workspaceRoot,{maximum:402653184}).total,capacity=extra=>{if(occupied()+extra>budget)fail('input vault plus raw/actual Git source-bank aggregate exceeds384MiB');};if(budget<=indexReserve)fail('input vault leaves no bounded Git source-bank storage');mkdirSync(root,{mode:0o700});run(['init','--quiet','--template=','--initial-branch=router-compiler-source-data','--object-format='+objectFormat]);
 // Each decode lease has the original64MiB occurrence bound. There is no
 // decoded compiler/proof/verdict cache; only current transfer bytes are shared.
 let cursor=0;while(cursor<descriptions.length){const lease=[];let bytes=0;while(cursor<descriptions.length&&bytes+descriptions[cursor].bytes<=67108864){const row=descriptions[cursor++];lease.push(row);bytes+=row.bytes;}if(!lease.length)fail('one source exceeds complete decode lease bound');withCompactSourceVault(model,reader=>{let at=0;while(at<lease.length){remaining();const batch=[];let rawBytes=0;while(at<lease.length&&batch.length<128){const row=lease[at];if(batch.length&&rawBytes+row.bytes>1048576)break;const raw=reader.read(row.path),blob=oid(raw,objectFormat);batch.push({...row,raw,blob});rawBytes+=raw.length;at++;}const packetBytes=rawBytes+batch.length*128+16;if(packetBytes>33587200)fail('complete fast-import transfer packet bound');capacity(rawBytes+Math.ceil(rawBytes/16384)*8+batch.length*256+65536+indexReserve);const pieces=[];for(const row of batch)pieces.push(Buffer.from('blob\ndata '+row.raw.length+'\n'),row.raw,Buffer.from('\n'));pieces.push(Buffer.from('done\n'));let packet=Buffer.concat(pieces);pieces.length=0;try{run(['fast-import','--quiet','--done','--depth=0','--max-pack-size=16777216'],{input:packet,maxBytes:1048576});packet=null;const actual=run(['cat-file','--batch'],{input:batch.map(row=>row.blob).join('\n')+'\n',encoding:null,maxBytes:rawBytes+batch.length*256+1024});verifyBatchOutput(actual,batch);for(const row of batch)rows.push({path:row.path,mode:row.mode,blob:row.blob,bytes:row.bytes});capacity(indexReserve);}finally{packet=null;for(const row of batch)row.raw=null;batch.length=0;}}return null;});}

 if(hot.length){const hotBytes=hot.reduce((sum,name)=>sum+available.get(name).bytes,0);capacity(hotBytes+65536);const actualBlobs=new Map(rows.map(row=>[row.path,row.blob])),hotRecords=hot.map(path=>available.get(path));let at=0;while(at<hotRecords.length){const group=[];let count=0;while(at<hotRecords.length&&count+hotRecords[at].bytes<=67108864){const row=hotRecords[at++];group.push(row);count+=row.bytes;}withCompactSourceVault(model,reader=>{for(const row of group){const actual=run(['cat-file','blob',actualBlobs.get(row.path)],{encoding:null,maxBytes:row.bytes+1});if(!actual.equals(reader.read(row.path)))fail('whole actual hot Git blob bytes differ');writeCompactSparseHotPhysicalFile(root,row.path,actual,row.mode);}return null;});}run(['update-index','--no-skip-worktree','-z','--stdin'],{input:hot.map(name=>name+'\0').join(''),maxBytes:1048576});}
 const expected=rows.map(row=>`${row.mode} ${row.blob} 0\t${row.path}`),actual=run(['ls-files','--stage','-z']).split('\0').filter(Boolean);if(JSON.stringify(actual)!==JSON.stringify(expected))fail('complete real source-bank index differs');const flags=run(['ls-files','-v','-z']).split('\0').filter(Boolean),wanted=rows.map(row=>(hotSet.has(row.path)?'H ':'S ')+row.path);if(JSON.stringify(flags)!==JSON.stringify(wanted))fail('complete real physical/sparse source flags differ');capacity(0);outputWatch=compactSparsePhysicalCensus(root,{retainBytes:true,maximum:402653184});close();return {root,compilerBankCommit:commit,originalApplicationHeadClaimed:false,originalSourceReceiptPath:available.has('translator-export/source-receipt.json')?'translator-export/source-receipt.json':null,files:rows.length,logicalSourceBytes:rows.reduce((sum,row)=>sum+row.bytes,0),hotFiles:hot.length,inputVaultBytes:inputWatch.total,sourceBankBytes:outputWatch.total,workspaceRoot,wholeWorkspaceBytes:occupied(),sourceStorageProfile:'readonly-source-profiles-full-compiler-bank-384MiB-v1',allLogicalSourcesRetained:true,sourceProfilesPhysicallyHot:false,nativeCodeRustSdkPhysical:true,aggregateBytes:inputWatch.total+outputWatch.total,objectFormat,sourceAuthority:false,semanticAuthority:false,compilerQualified:false,receivingQualified:false,nativeCompilerExecuted:false};
 },[close]);}finally{releaseCompactSparsePhysicalCensus(inputWatch);releaseCompactSparsePhysicalCensus(outputWatch);rows.length=0;if(model){model.volumes.length=0;model.metadata.files.length=0;model.metadata.objects.length=0;}/* On failure retain created source bank for caller-owned guarded cleanup. */}
 })),[remaining]);
 });
