/** One operation owns one actual spawned child. No PID from disk is signalled. */
import {spawn} from 'node:child_process';
import {isAbsolute} from 'node:path';
import {constants} from 'node:os';
import {clientFailure} from './client-files.mjs';
function boundedInteger(value/*:unknown*/,maximum/*:number*/,label/*:string*/)/*:number*/{if(!Number.isSafeInteger(value)||Number(value)<1||Number(value)>maximum)throw clientFailure(label+' exceeds its finite bound');return Number(value);}
function launchInput(context/*:any*/)/*:any*/{
 const {command,args=[],env={},cwd}=context;if(typeof command!=='string'||!command||command.includes('\0')||Buffer.byteLength(command)>4096)throw clientFailure('Client command requires a bounded executable name');

 const environment/*:Record<string,string>*/=Object.create(null);let bytes=0;const entries=Object.entries(env);if(entries.length>4096)throw clientFailure('Client environment exceeds its entry bound');
 for(const [key,value]of entries){if(!key||key.includes('=')||key.includes('\0')||typeof value!=='string'||value.includes('\0'))throw clientFailure('Client environment requires NUL-free string entries');bytes+=Buffer.byteLength(key)+Buffer.byteLength(value)+2;if(bytes>2097152)throw clientFailure('Client environment exceeds its byte bound');environment[key]=value;}
 return {command,args:[...args],cwd,env:environment,timeout:boundedInteger(context.timeoutMs??60000,86400000,'Client deadline'),limit:boundedInteger(context.maxOutputBytes??8388608,8388608,'Client output cap'),interactive:context.interactive===true};
}
export async function runOwnedClientProcess(context/*:any*/)/*:Promise<any>*/{

 let deadline/*:ReturnType<typeof setTimeout>|undefined*/,grace/*:ReturnType<typeof setTimeout>|undefined*/,drain/*:ReturnType<typeof setTimeout>|undefined*/,lastResort/*:ReturnType<typeof setTimeout>|undefined*/;

 const finish=()/*:void*/=>{if(settled)return;settled=true;complete();};

  failure??=clientFailure(message);if(settled)return;if(!exited){try{child.kill(requestedSignal);}catch{}if(!grace)grace=setTimeout(()=>{if(!exited)try{child.kill('SIGKILL');}catch{}},3000);}
  if(!lastResort)lastResort=setTimeout(()=>{child.stdout?.destroy();child.stderr?.destroy();child.unref();finish();},5000);
 };
 const cancel=()/*:void*/=>stop('Client launch cancelled'),interrupt=()/*:void*/=>stop('Client launch interrupted','SIGINT'),terminate=()/*:void*/=>stop('Client launch terminated','SIGTERM');
 const collect=(destination/*:Buffer[]*/,chunk/*:Buffer*/,channel/*:'stdout'|'stderr'*/)/*:void*/=>{
  if(channel==='stdout')stdoutBytes+=chunk.length;else stderrBytes+=chunk.length;
  if((channel==='stdout'?stdoutBytes:stderrBytes)>input.limit){stop('Client '+channel+' exceeds its output cap');return;}destination.push(Buffer.from(chunk));
 };
 child.stdout?.on('data',(chunk/*:Buffer*/)=>collect(stdout,chunk,'stdout'));child.stderr?.on('data',(chunk/*:Buffer*/)=>collect(stderr,chunk,'stderr'));
 child.stdout?.on('error',()=>stop('Client stdout could not be read'));child.stderr?.on('error',()=>stop('Client stderr could not be read'));
 child.once('error',(error/*:NodeJS.ErrnoException*/)=>{failure=clientFailure(error.code==='ENOENT'?'Client executable is not installed or not on PATH':'Client executable could not be started');});

 child.once('close',(status/*:number|null*/,reason/*:NodeJS.Signals|null*/)=>{if(code===null)code=status;if(exitSignal===null)exitSignal=reason;finish();});

  await finished;const result={client_exit_code:code??(exitSignal?128+(Object( constants.signals )[exitSignal]??0):1),signal:exitSignal,stdout:Buffer.concat(stdout).toString('utf8'),stderr:Buffer.concat(stderr).toString('utf8')};
  if(failure){Object.assign(failure,{data:result});throw failure;}return result;
 }finally{clearTimeout(deadline);clearTimeout(grace);clearTimeout(drain);clearTimeout(lastResort);signal?.removeEventListener('abort',cancel);if(context.forwardSignals===true){process.off('SIGINT',interrupt);process.off('SIGTERM',terminate);}child.stdout?.destroy();child.stderr?.destroy();}
}
