/** Bounded verified profile copies using the native version-one backup manifest shape. */
import {open,lstat,mkdir,chmod,readdir,copyFile,rename,rm,statfs} from 'node:fs/promises';import {constants} from 'node:fs';import {createHash,randomUUID} from 'node:crypto';import {join,relative,resolve,isAbsolute,dirname} from 'node:path';
import {clientFailure} from './client-files.mjs';import {requireNativePath,readNativeText} from './operation-input.mjs';import {atomicWrite} from './storage.mjs';
const hash=(bytes/*:any*/)/*:string*/=>createHash('sha256').update(bytes).digest('hex'),disposable=new Set(['cache','Cache','CachedData','node_modules']);
function credential(name/*:string*/)/*:boolean*/{name=name.toLowerCase();return ['.claude.json','auth.json','oauth_creds.json','.credentials.json','credentials.json','.env'].includes(name)||['secret','token','keychain'].some(word=>name.includes(word));}
async function digest(path/*:string*/)/*:Promise<any>*/{
 const file=await open(path,constants.O_RDONLY|(constants.O_NOFOLLOW??0));try{const before=await file.stat();if(!before.isFile()||before.size>67108864)throw clientFailure('Profile file exceeds its 64 MiB backup bound');const value=createHash('sha256'),buffer=Buffer.alloc(65536);let size=0;for(;;){const part=await file.read(buffer,0,buffer.length,null);if(!part.bytesRead)break;size+=part.bytesRead;if(size>67108864)throw clientFailure('Profile file grew beyond its backup bound');value.update(buffer.subarray(0,part.bytesRead));}const after=await file.stat();if(size!==before.size||after.size!==before.size||after.mtimeMs!==before.mtimeMs||after.ctimeMs!==before.ctimeMs)throw clientFailure('Profile file changed during backup hashing');return {sha256:value.digest('hex'),size,mode:before.mode&0o7777};}finally{await file.close();}
}
export async function snapshotClientProfile(root/*:string*/,selection/*:any*/={client:'claude',scope:'router',store:'home'},options/*:any*/={})/*:Promise<any>*/{
 selection={client:selection.client,scope:selection.scope,store:selection.store};await requireNativePath(root);const entries/*:any[]*/=[];let count=0,total=0,omitted=0;
 const visit=async(path/*:string*/,depth/*:number*/)/*:Promise<void>*/=>{if(++count>10000||depth>32)throw clientFailure('Profile backup exceeds its entry/depth bound');const name=relative(root,path).split('\\').join('/'),item=await lstat(path);if(!options.allEntries&&name&&name.split('/').some(part=>disposable.has(part)))return;if(!options.allEntries&&item.isFile()&&credential(name?name.split('/').at(-1):path.split('/').at(-1))){omitted++;return;}if(item.isSocket())return;if(item.isSymbolicLink())throw clientFailure('Profile symlink requires the remaining qualified link-copy boundary');if(!item.isFile()&&!item.isDirectory())throw clientFailure('Unsupported special profile entry');if(item.isFile()&&['-wal','-shm','-journal'].some(suffix=>name.endsWith(suffix)))throw clientFailure('Live SQLite sidecar requires the client to close before backup');const row={...selection,path:name,kind:item.isDirectory()?'directory':'file',size:0,sha256:null,link:null,backup_link:null,mode:item.mode&0o7777};if(item.isFile()){Object.assign(row,await digest(path));total+=row.size;if(total>67108864)throw clientFailure('Profile backup exceeds its total 64 MiB bound');}entries.push(row);if(item.isDirectory()){const names=await readdir(path,{encoding:'buffer'});names.sort(Buffer.compare);for(const bytes of names){let child;try{child=new TextDecoder('utf8',{fatal:true}).decode(bytes);}catch{throw clientFailure('Non-UTF8 profile filename is unsupported');}await visit(join(path,child),depth+1);}}};
 await visit(root,0);return {entries,total,omitted};
}
function selectedPath(root/*:string*/,path/*:string*/)/*:string*/{if(typeof path!=='string'||isAbsolute(path)||path.includes('\\')||path.split('/').some(word=>word==='.'||word==='..')||path.startsWith('/'))throw clientFailure('Backup entry requires a safe relative path');return join(root,path);}
export async function verifyClientProfileBackup(directory/*:string*/)/*:Promise<any>*/{
 const text=await readNativeText(join(directory,'manifest.json'),{maxBytes:8388608}),checksum=await readNativeText(join(directory,'manifest.sha256'),{maxBytes:128});if(hash(Buffer.from(text))!==checksum)throw clientFailure('Profile backup manifest checksum differs');let manifest;try{manifest=JSON.parse(text);}catch{throw clientFailure('Profile backup manifest is invalid JSON');}
 if(manifest.version!==1||typeof manifest.id!=='string'||!/^[a-f0-9]{32}$/u.test(manifest.id)||!Array.isArray(manifest.entries)||manifest.entries.length>10000||manifest.stores?.length!==1||manifest.credentials_included!==false)throw clientFailure('Profile backup manifest shape differs');const store=manifest.stores[0];if(!['client','scope','store'].every(key=>typeof store[key]==='string'&&/^[a-z0-9-]+$/u.test(store[key])))throw clientFailure('Profile backup store labels differ');const data=join(directory,'data',store.client,store.scope,store.store),seen=new Set();
 for(const row of manifest.entries){if(row.client!==store.client||row.scope!==store.scope||row.store!==store.store||seen.has(row.path))throw clientFailure('Profile backup entries differ');seen.add(row.path);const path=selectedPath(data,row.path);await requireNativePath(path);const item=await lstat(path);if(row.kind==='file'){const actual=await digest(path);if(actual.size!==row.size||actual.sha256!==row.sha256||actual.mode!==row.mode)throw clientFailure('Profile backup file verification failed');}else if(row.kind!=='directory'||!item.isDirectory()||(item.mode&0o7777)!==row.mode)throw clientFailure('Profile backup directory verification failed');}
 const actual=await snapshotClientProfile(data,{client:store.client,scope:store.scope,store:store.store},{allEntries:true});if(JSON.stringify(actual.entries)!==JSON.stringify(manifest.entries))throw clientFailure('Profile backup contains missing or unlisted entries');return manifest;
}
export async function createVerifiedClientProfileBackup(profile/*:string*/,backupRoot/*:string*/,selection/*:any*/={client:'claude',scope:'router',store:'home'})/*:Promise<any>*/{
 profile=resolve(profile);backupRoot=resolve(backupRoot);if(backupRoot===profile||backupRoot.startsWith(profile+'/'))throw clientFailure('Backup destination is inside the selected profile');if(!['client','scope','store'].every(key=>typeof selection[key]==='string'&&/^[a-z0-9-]+$/u.test(selection[key])))throw clientFailure('Profile backup selection requires fixed labels');await requireNativePath(backupRoot);await mkdir(backupRoot,{recursive:true,mode:0o700});await chmod(backupRoot,0o700);
 const before=await snapshotClientProfile(profile,selection),space=await statfs(backupRoot,{bigint:true});if(space.bavail*space.bsize<BigInt(before.total+1048576))throw clientFailure('Profile backup destination lacks space');const id=randomUUID().replaceAll('-',''),temporary=join(backupRoot,'.incomplete-'+id),destination=join(backupRoot,id);await mkdir(temporary,{mode:0o700});const original=await lstat(temporary),data=join(temporary,'data',selection.client,selection.scope,selection.store);
 try{
  for(const row of before.entries){const source=selectedPath(profile,row.path),output=selectedPath(data,row.path);await requireNativePath(source);if(row.kind==='directory'){await mkdir(output,{recursive:true,mode:0o700});}else{await mkdir(dirname(output),{recursive:true,mode:0o700});await copyFile(source,output,constants.COPYFILE_FICLONE);await chmod(output,row.mode);const actual=await digest(output);if(actual.sha256!==row.sha256||actual.size!==row.size)throw clientFailure('Profile changed while copying backup');}}
  for(const row of [...before.entries].reverse())if(row.kind==='directory')await chmod(selectedPath(data,row.path),row.mode);
  if(JSON.stringify(await snapshotClientProfile(profile,selection))!==JSON.stringify(before))throw clientFailure('Profile changed during complete backup');const manifest={version:1,id,created:new Date().toISOString(),credentials_included:false,stores:[{...selection,source:profile}],entries:before.entries,unavailable:[],omitted_credentials:before.omitted},text=JSON.stringify(manifest,null,2);await atomicWrite(join(temporary,'manifest.json'),text);await atomicWrite(join(temporary,'manifest.sha256'),hash(Buffer.from(text)));await verifyClientProfileBackup(temporary);await rename(temporary,destination);return {id,directory:destination,snapshot:before};
 }catch(error){const current=await lstat(temporary).catch(()=>null);if(current&&current.dev===original.dev&&current.ino===original.ino&&!current.isSymbolicLink())await rm(temporary,{recursive:true,force:true,maxRetries:2,retryDelay:100});throw error;}
}
