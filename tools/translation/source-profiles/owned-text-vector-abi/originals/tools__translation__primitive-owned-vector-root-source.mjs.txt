export function proposePrimitiveOwnedVectorRootSource(source,{sourcePath='src/lib.rs',runtimeRoot='runtime',declarationRoot=runtimeRoot}={}){
 if(typeof source!=='string'||Buffer.byteLength(source)>2097152||![sourcePath,runtimeRoot,declarationRoot].every(safe)||!sourcePath.endsWith('.rs')||['src/translation_primitive_vectors_core.rs','src/translation_primitive_sequences_core.rs'].includes(sourcePath))fail('complete bounded original root and safe receiving paths required');
