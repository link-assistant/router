// Whole active receiving-source observations. These records are DATA only;
// the actual parser/checker must still bind the complete caller and native ABI.
import {withFreshPrimitiveSequenceInputs} from './fresh-primitive-sequence-inputs.mjs';
import {captureSourceModuleViews,originalModuleViewItems} from './source-module-views.mjs';
import {transportConcreteSyntaxSource,transportConcreteSyntaxMeta} from './concrete-syntax-meta.mjs';
import {primitiveSequenceContract,PRIMITIVE_SEQUENCE_CONTRACTS} from './primitive-sequence-contracts.mjs';
const fail=message=>{throw new TypeError('Primitive sequence source binding: '+message);};
export function observePrimitiveSequenceCoreSources(files,{runtimeRoot,declarationRoot=runtimeRoot}){
 return withFreshPrimitiveSequenceInputs(frame=>{
  const root=files.get('src/lib.rs');if(typeof root!=='string')fail('complete actual receiving root required');const concrete=transportConcreteSyntaxSource(root,{language:'Rust'});if(transportConcreteSyntaxMeta(concrete.meta).source!==root)fail('complete receiving root CST inverse differs');const capture=captureSourceModuleViews({source:root,sourcePath:'src/lib.rs'}),items=originalModuleViewItems(capture.views[0],root),edges=items.filter(item=>item.term==='mod'&&item.name==='translation_primitive_sequences_core');if(edges.length!==1||root.slice(edges[0].start,edges[0].end).trim()!=='pub mod translation_primitive_sequences_core;'||items.some(item=>item.term==='use'&&item.tokens.some(token=>token.text==='translation_primitive_sequences_core')))fail('one actual unambiguous public native edge required');
  const records=[{path:'src/lib.rs',source:root,concreteSyntax:concrete.meta,inverse:root}],native=frame.sources.find(row=>row.language==='Rust');for(const row of frame.sources){if(row.path==='primitive-sequence-contracts.mjs')continue;const paths=row.language==='Rust'?[row.path]:row.language==='TypeScript'?[declarationRoot+'/'+row.path,...(runtimeRoot===declarationRoot?[]:[runtimeRoot+'/'+row.path])]:[runtimeRoot+'/'+row.path,...(runtimeRoot===declarationRoot?[]:[declarationRoot+'/'+row.path])];for(const path of paths){if(files.get(path)!==row.source)fail('complete actual library/runtime/declaration differs: '+path);records.push({path,source:row.source,concreteSyntax:row.concreteSyntax,inverse:row.source});}}
  const nativeCapture=captureSourceModuleViews({source:native.source,sourcePath:native.path}),declarations=originalModuleViewItems(nativeCapture.views[0],native.source),functions=declarations.filter(item=>item.term==='fn'),publicFunctions=functions.filter(item=>{const at=item.tokens.findIndex(token=>token.text==='fn');return at>0&&item.tokens[at-1].text==='pub';}),helpers=functions.filter(item=>!publicFunctions.includes(item));
  if(publicFunctions.length!==33||PRIMITIVE_SEQUENCE_CONTRACTS.length!==33||publicFunctions.some(item=>!primitiveSequenceContract(item.name))||helpers.length!==2||!['primitive_sequence_length','primitive_sequence_get'].every(name=>helpers.filter(item=>item.name===name).length===1))fail('complete actual public and private native operation census differs');
  return {schema:'primitive-sequence-core-source-observation-v1',records,nativeSource:native,publicFunctions:publicFunctions.map(item=>({name:item.name,span:{start:item.start,end:item.end},original:native.source.slice(item.start,item.end)})),sourceAuthority:false,semanticAuthority:false,namespaceAuthority:false,runtimeAuthority:false,privateCallerBindingQualified:false,nativeRustGrammarChecked:false,receivingExecuted:false,crossInvocationCache:false};
 });
}
