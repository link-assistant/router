// Independent division/rem origin DATA cases; no synthetic source authority.
import test from 'node:test';
import assert from 'node:assert/strict';
import {collectOpaqueBodyNumericOriginData} from '../opaque-body-numeric-origins-20261102.mjs';
function fixture(op,symbol){const source='pub fn calculate(a:i64,b:i64)->i64{a '+symbol+' b}',type={kind:'fixed',rustPrimitiveName:'i64',bits:64,signed:true},left=source.indexOf('{a ')+1,right=source.lastIndexOf(' b}')+1,proof={schema:'original-checked-executable-source-v1',source:'rust/src/calculate.rs',qualifiedName:'calculate.calculate',span:{start:0,end:source.length},checkedSpan:{start:0,end:source.length},declaration:{body:{k:'binary',op,type,semantics:'checked',rounding:'trunc',byZero:'abort',left:{k:'var',type,span:{start:left,end:left+1}},right:{k:'var',type,span:{start:right,end:right+1}}}}},row={source:proof.source,qualifiedName:proof.qualifiedName,checkedSource:()=>proof,targetBinding:()=>({schema:'original-checked-project-target-binding-v1',source:proof.source,qualifiedName:proof.qualifiedName,javascriptName:'calculate_calculate'})};return {source,proof,fresh:{executableDeclarations:[row],meta:{executable:[{source:proof.source,name:proof.qualifiedName,term:'fn'}]}}};}
test('division and remainder require exact checked truncating aborting native operator origins',()=>{for(const [op,symbol]of [['div','/'],['rem','%']]){const value=fixture(op,symbol),rows=collectOpaqueBodyNumericOriginData(value.fresh,()=>value.source);assert.equal(rows.length,1);assert.equal(rows[0].kind,'division');assert.equal(rows[0].value,symbol);assert.equal(rows[0].nativeToken,symbol);assert.deepEqual(rows[0].domain,{name:'i64',bits:64,signed:true});assert.equal(rows[0].sourceAuthority,false);assert.equal(rows[0].semanticAuthority,false);}});
test('nontruncating total float wrapping unresolved and mismatched division domains remain unsupported',()=>{for(const change of [value=>{value.proof.declaration.body.rounding='floor';},value=>{value.proof.declaration.body.byZero='total';},value=>{value.proof.declaration.body.semantics='wrapping';},value=>{value.proof.declaration.body.type={kind:'float'};},value=>{value.proof.declaration.body.type={...value.proof.declaration.body.type,rustBorrowingUnresolved:true};},value=>{value.source=value.source.replace('/','%');}]){const value=fixture('div','/');change(value);assert.deepEqual(collectOpaqueBodyNumericOriginData(value.fresh,()=>value.source),[]);}});
