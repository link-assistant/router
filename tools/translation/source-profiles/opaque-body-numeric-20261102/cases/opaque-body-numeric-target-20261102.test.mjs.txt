// Native source-shape candidate units. Every execution UNRUN.
import test from 'node:test';
import assert from 'node:assert/strict';
import {ownershipCompiler} from '../js-to-rust/vendor/module-syntax.mjs';
import {opaqueCheckedDivisionCallParts,opaqueDivisionProjectionText,opaqueNumericOriginKind} from '../opaque-body-numeric-target-20261102.mjs';
const limits={timeout:60000};
function parts(source){const ts=ownershipCompiler(),file=ts.createSourceFile('division.mjs',source,ts.ScriptTarget.Latest,true,ts.ScriptKind.JS),rows=[];assert.equal(file.parseDiagnostics.length,0);function visit(node){const value=opaqueCheckedDivisionCallParts(ts,node,source);if(value)rows.push(value);ts.forEachChild(node,visit);}visit(file);return rows;}
const examples=[['/',false,'attempt to divide by zero','attempt to divide with overflow'],['%',true,'attempt to calculate the remainder with a divisor of zero','attempt to calculate the remainder with overflow']];
test('each complete checked call exposes three independently expected projections for one native operation',limits,()=>{for(const [symbol,marker,zero,overflow]of examples){const source='function value(a,b){return ml_divide(a,b,"trunc",'+JSON.stringify(zero)+','+marker+',[(-128n),127n,'+JSON.stringify(overflow)+']);}',rows=parts(source);assert.equal(rows.length,1);assert.equal(rows[0].operation.target,symbol);assert.deepEqual(rows[0].cells.map(row=>row.kind),['division-marker','division-zero-message','division-overflow-message']);assert.ok(rows[0].cells.every(row=>row.operation.target===symbol));assert.deepEqual(opaqueDivisionProjectionText(symbol),{marker,zero,overflow,sourceAuthority:false,semanticAuthority:false});for(const cell of rows[0].cells)assert.equal(opaqueNumericOriginKind(cell.kind),'division');assert.equal(rows[0].semanticAuthority,false);}assert.throws(()=>opaqueDivisionProjectionText('+'));});
test('wrong helper rounding envelope or unknown message cannot produce a selected call candidate',limits,()=>{for(const source of ['function f(a,b){return other(a,b,"trunc","attempt to divide by zero",false,[0n,255n,"attempt to divide with overflow"]);}','function f(a,b){return ml_divide(a,b,"floor","attempt to divide by zero",false,[0n,255n,"attempt to divide with overflow"]);}','function f(a,b){return ml_divide(a,b,"trunc",null,false,[0n,255n,"attempt to divide with overflow"]);}','function f(a,b){return ml_divide(a,b,"trunc","attempt to divide by zero",false);}'])assert.deepEqual(parts(source),[]);});
