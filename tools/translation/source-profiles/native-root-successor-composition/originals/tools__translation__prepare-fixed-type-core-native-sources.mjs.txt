import {FIXED_TYPE_CORE_NATIVE_SOURCE,FIXED_TYPE_CORE_ALIASES} from './fixed-type-core-source.mjs';
const ROOT=fileURLToPath(new URL('../../',import.meta.url)),ARCHIVE='tools/translation/source-profiles/fixed-type-core-source/complete-native-original.json.gz',PIN='df41589cde056c5b3c0402ea329f9de851f7d00a89c9c546dfb463c154554d89';
const hash=value=>createHash('sha256').update(value).digest('hex'),need=(value,message)=>{if(!value)throw new TypeError('Fixed type core native source: '+message);};
export function prepareFixedTypeCoreNativeSources({root=ROOT}={}){
 root=resolve(root);const packed=readFileSync(resolve(root,ARCHIVE));need(packed.length<=1048576&&hash(packed)===PIN,'complete native template archive differs');const original=JSON.parse(gunzipSync(packed,{maxOutputLength:2097152}).toString('utf8'));need(original.schema==='fixed-type-core-native-original-v1'&&original.template.path==='rust/src/translation_core/primitives.rs'&&original.root.path==='rust/src/lib.rs'&&hash(original.template.source)===original.template.sha256&&hash(original.root.source)===original.root.sha256,'complete original native sources differ');
 const declarations=sourceItems(original.template.source).filter(item=>item.term==='type'&&item.name==='Str');need(declarations.length===1,'one complete original Str alias template required');const tokens=declarations[0].tokens.filter(token=>token.type!=='comment'),template=original.template.source.slice(tokens[0].start,tokens.at(-1).end);need(template===original.declaration&&JSON.stringify(original.contracts)===JSON.stringify(FIXED_TYPE_CORE_ALIASES.slice(0,-2))&&JSON.stringify(FIXED_TYPE_CORE_ALIASES.slice(-2))===JSON.stringify([{name:'Usize',primitive:'usize'},{name:'Isize',primitive:'isize'}]),'actual complete native alias template/contracts required');
 const rootBefore=original.root.source,anchors=sourceItems(rootBefore).filter(item=>item.term==='mod'&&item.name==='translation_integer_core');need(anchors.length===1&&!sourceItems(rootBefore).some(item=>item.term==='mod'&&item.name==='translation_fixed_types_core'),'one original native module anchor and absent target required');const anchor='pub mod translation_integer_core;';need(rootBefore.split(anchor).length===2,'complete native root anchor required');
 const prefix='//! Explicit scalar types for the owned Rust/JavaScript/TypeScript interface.\n//! Standard library internals remain intentionally untranslated.\n\n',libraryAfter=prefix+FIXED_TYPE_CORE_ALIASES.map(row=>template.replace(' Str ',' '+row.name+' ').replace('::str;','::'+row.primitive+';')).join('\n')+'\n',libraryLanguage=libraryAfter.replaceAll('pub type ','type '),rootLanguage=rootBefore.replace(anchor,anchor+'\nmod translation_fixed_types_core;'),rootAfter=rootBefore.replace(anchor,anchor+'\npub mod translation_fixed_types_core;');need(libraryAfter===FIXED_TYPE_CORE_NATIVE_SOURCE,'whole native library source differs');
 const run=createRulePipeline([{phase:'language',rules:[{id:'rust:copy-actual-complete-scalar-alias-template-and-native-root-source',apply:()=>({library:libraryLanguage,root:rootLanguage})}]},{phase:'repository',rules:[{id:'router:publish-only-explicit-qualified-scalar-core-types',apply:()=>({library:libraryAfter,root:rootAfter})}]}]).run({library:original.template.source,root:rootBefore});
 const files=[{path:'rust/src/translation_fixed_types_core.rs',sourceOriginPath:original.template.path,targetBeforeExists:false,targetBefore:null,before:original.template.source,language:libraryLanguage,after:libraryAfter},{path:original.root.path,sourceOriginPath:original.root.path,targetBeforeExists:true,targetBefore:rootBefore,before:rootBefore,language:rootLanguage,after:rootAfter}].map(row=>{const stages=transportConcreteSourceEquations([{before:row.before,after:row.language,language:'Rust'},{before:row.language,after:row.after,language:'Rust'}]).map((stage,index)=>({...stage,phase:index?'repository':'language'})),current=readFileSync(resolve(root,row.path),'utf8'),history=composeConcreteSourceHistory({path:row.path,before:row.before,after:row.after,language:'Rust',laterPlans:row.path===original.root.path?[preparePrimitiveSequenceRootActivation({root})]:[]});need((current===row.after||current===history.currentAfter)&&stages[0].inverse===row.before&&stages[1].inverse===row.language,'complete native current source/CST inverse differs');return {...row,current,...history,stages,inverse:row.before,nativeRustGrammarChecked:false,sourceAuthority:false,semanticAuthority:false,rustExecuted:false,receivingExecuted:false};});
 const actual=sourceItems(libraryAfter).filter(item=>item.term==='type');need(actual.length===14&&FIXED_TYPE_CORE_ALIASES.every(row=>actual.filter(item=>item.name===row.name).length===1),'all explicit native scalar aliases required');need(readFileSync(resolve(root,original.template.path),'utf8')===original.template.source&&readFileSync(resolve(root,ARCHIVE)).equals(packed)&&files.every(row=>readFileSync(resolve(root,row.path),'utf8')===row.current),'complete native inputs changed during construction');return {schema:'fixed-type-core-native-source-construction-v1',files,trace:run.trace,nativeRustGrammarChecked:false,sourceAuthority:false,semanticAuthority:false,namespaceAuthority:false,rustExecuted:false,receivingExecuted:false};
}
