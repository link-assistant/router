const ROOT=fileURLToPath(new URL('../../',import.meta.url));
export function prepareTextBuilderCarrierReturnSources({root=ROOT}={}){
 return withNativeSubstitutionOperation(()=>prepareNativeSubstitutionCopy({root,archive:'tools/translation/source-profiles/text-builder-carrier-return-source/complete-originals-and-rules.json.gz',pin:'3c2d7f433c1fd8ff31f20a1ad509bc2a140519ccc6a657795edd314e37dff486',id:'router:text-builder-carrier-return-source-complete-source-substitutions-v1'}));
}
if(process.argv[1]&&resolve(process.argv[1])===fileURLToPath(import.meta.url)){
 const args=process.argv.slice(2);if(args.length!==1||!['--emit','--write','--check'].includes(args[0]))throw new TypeError('Use --emit|--write|--check');const plan=prepareTextBuilderCarrierReturnSources();if(args[0]==='--write')writeNativeSubstitutionCopy(plan,{root:ROOT});if(args[0]==='--check'&&nativeSubstitutionCopyHasChanges(plan))throw new TypeError('Text builder carrier return source differs');console.log(JSON.stringify(args[0]==='--emit'?plan:{schema:plan.schema,files:plan.files.map(row=>row.path),receivingExecuted:false}));
