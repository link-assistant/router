});

 let called=false;const value={};Object.defineProperty(value,'source',{enumerable:true,get(){called=true;return 'foreign';}});assert.throws(()=>accountForwardSourceProposalData(value),/without accessors/);assert.equal(called,false);assert.throws(()=>accountForwardSourceProposalData(new Proxy({},{})),/ordinary source DATA/);const cyclic={};cyclic.self=cyclic;assert.throws(()=>accountForwardSourceProposalData(cyclic),/cyclic/);assert.throws(()=>accountForwardSourceProposalData({number:Infinity}),/finite/);
