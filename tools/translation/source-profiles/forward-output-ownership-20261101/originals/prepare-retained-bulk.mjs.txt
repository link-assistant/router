// Deterministic proposal/checker; original compiler semantics remain in the
// complete source-derived modules. This tool runs under stock host intrinsics.
import {readFileSync} from 'node:fs';
import {resolve,dirname,join} from 'node:path';
import {fileURLToPath} from 'node:url';
import {createHash} from 'node:crypto';
import {createRulePipeline,exactSubstitutionRule} from './rule-pipeline.mjs';
const LANGUAGE_EDITS = [
  {
    "id": "storage:private-plan-associations",
    "before": "const plans=new WeakMap();",
    "after": "const plans=createPrivateOutputPlanStore();"
  },
  {
    "id": "storage:private-text-values",
    "before": "const expected = new Map();",
    "after": "const expected = createPrivateOutputTextMap();"
  },
  {
    "id": "storage:one-text-at-byte-census",
    "before": "const expectedBytes=[...expected.values()].reduce((sum,content)=>sum+Buffer.byteLength(content),0);",
    "after": "let expectedBytes=0;for(const content of expected.values())expectedBytes+=Buffer.byteLength(content);"
  },
  {
    "id": "storage:one-text-at-output-hash",
    "before": "const outputs=[...expected].map(([path,content])=>Object.freeze({path,bytes:Buffer.byteLength(content),sha256:sha256(content)}));",
    "after": "const outputs=[];for(const [path,content] of expected)outputs.push(Object.freeze({path,bytes:Buffer.byteLength(content),sha256:sha256(content)}));"
  }
];
const REPOSITORY_EDITS = [
  {
    "id": "repository:private-output-helper",
    "before": "import {planCompressedProjectMetaStorage as planProjectMetaStorage} from './project-meta-compressed-storage.mjs';",
    "after": "import {planCompressedProjectMetaStorage as planProjectMetaStorage} from './project-meta-compressed-storage.mjs';\nimport {createPrivateOutputPlanStore,createPrivateOutputTextMap} from './private-output-text.mjs';"
  }
];
const MATRIX_EDITS = [
  {
    "id": "repository:retained-profile-planner",
    "before": "from './bulk.mjs';",
    "after": "from './retained-bulk.mjs';"
  }
];
const HEADER = "// Generated by prepare-retained-bulk.mjs; explicit stock-private-collection CLI contract.\n// No old Map/WeakMap observer equivalence or new semantic/source authority.\n";
const hash=text=>createHash('sha256').update(text).digest('hex');
function derive(source,languageEdits,repositoryEdits){
  if(typeof source!=='string'||Buffer.byteLength(source)>1048576)throw new TypeError('Complete bounded original bulk source required');
  const rules=rows=>rows.map(row=>exactSubstitutionRule(row.id,row.before,row.after,'Original bulk unique anchor changed: '+row.id));
  const runner=createRulePipeline([{phase:'language',rules:rules(languageEdits)},{phase:'repository',rules:[...rules(repositoryEdits),{id:'repository:closed-derived-module-header',apply:text=>HEADER+text}]}]);
  const result=runner.run(source);let undo=result.value;if(!undo.startsWith(HEADER))throw Error('Derived module header changed');undo=undo.slice(HEADER.length);
  const all=[...languageEdits,...repositoryEdits];for(let at=all.length-1;at>=0;at--){const row=all[at];if(undo.split(row.after).length!==2)throw Error('Derived full undo anchor changed');undo=undo.replace(row.after,row.before);}
  if(undo!==source)throw Error('Derived complete source undo changed');return {text:result.value,sourceSha256:hash(source),sha256:hash(result.value),trace:result.trace,wholeUndo:true};
}
export function deriveRetainedBulkSources({bulk,matrix}){
  return {schema:'private-output-retention-source-proposal-v1',bulk:derive(bulk,LANGUAGE_EDITS,REPOSITORY_EDITS),matrix:derive(matrix,[],MATRIX_EDITS),authority:{semantic:false,source:false,owner:false}};
}
if(process.argv[1]&&resolve(process.argv[1])===fileURLToPath(import.meta.url)){
  let root=resolve(dirname(fileURLToPath(import.meta.url)),'../..'),emit=false;const args=process.argv.slice(2);
  for(let at=0;at<args.length;at++){if(args[at]==='--root'&&args[at+1])root=resolve(args[++at]);else if(args[at]==='--emit')emit=true;else if(args[at]!=='--check')throw Error('Usage: prepare-retained-bulk.mjs [--root ROOT] [--check|--emit]');}
  const original=name=>{const bytes=readFileSync(join(root,'tools/translation',name));if(bytes.length>1048576)throw Error('Original module byte bound');const text=bytes.toString('utf8');if(!Buffer.from(text,'utf8').equals(bytes))throw Error('Original module UTF8 changed');return text;};
  const proposal=deriveRetainedBulkSources({bulk:original('bulk.mjs'),matrix:original('matrix-bulk.mjs')});
  if(emit)console.log(JSON.stringify(proposal));else{if(original('retained-bulk.mjs')!==proposal.bulk.text||original('retained-matrix-bulk.mjs')!==proposal.matrix.text)throw Error('Retained modules are stale; emit and review a new complete proposal');console.log(JSON.stringify({schema:proposal.schema,bulk:proposal.bulk.sha256,matrix:proposal.matrix.sha256,trace:{bulk:proposal.bulk.trace,matrix:proposal.matrix.trace},wholeUndo:true,checked:true}));}
}
