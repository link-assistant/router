// Explicit owned scalar aliases use the existing paired-library boundary.
// Actual root/module/scope identity is rejoined; public alias DATA is no permit.
import {FIXED_TYPE_CORE_NATIVE_SOURCE,FIXED_TYPE_CORE_ALIASES} from './fixed-type-core-source.mjs';
import {selectOriginalFixedTypeCore} from './fixed-type-core-interface.mjs';
import {matchesParserFacts} from './matches-macro-context.mjs';
import {originalSourceModuleItems} from './source-module-views.mjs';
import {rustFixedType,BOOL} from './checked-types.mjs';
import {transportConcreteSyntaxSource,transportConcreteSyntaxMeta} from './concrete-syntax-meta.mjs';
import {createRulePipeline} from './rule-pipeline.mjs';
import {unsupported} from './vendor/meta-language/translation/diagnostics.js';
export function parseRouterCoreFixedType(parser){
 const cursor=parser.cursor,words=['crate','::','translation_fixed_types_core','::'];if(!words.every((word,at)=>cursor.is(word,at)))return null;const token=cursor.peek(4),alias=token?.kind==='identifier'?FIXED_TYPE_CORE_ALIASES.find(row=>row.name===token.value):null;if(!alias)return null;
 const location={start:cursor.peek().start,end:token.end};if([0,2].some(at=>cursor.peek(at).kind!=='identifier')||cursor.is('<',5)||cursor.is('::',5))throw unsupported('owned primitive type','complete nongeneric qualified fixed type required',location);
 const before=matchesParserFacts(parser),tokens=cursor.tokens,start=cursor.index,actual=tokens.slice(start,start+5),select=()=>selectOriginalFixedTypeCore({modules:before.facts.modules,sourceScopes:before.facts.evidence.scopes,compilationProfile:before.facts.compilationProfile});let selected,run,nativeModule,owner,view,primitive;
 try{selected=select();nativeModule=before.facts.modules.find(module=>module.file===selected.file&&module.path.join('::')==='translation_fixed_types_core');const owners=originalSourceModuleItems(nativeModule).filter(item=>item.term==='type'&&item.name===token.value);if(owners.length!==1)throw new TypeError('Actual original scalar alias declaration required');owner=owners[0];view=nativeModule.sourceView;const words=owner.tokens.filter(row=>row.type!=='comment').map(row=>row.text);primitive=words[9];if(words.length!==11||!['pub','type',token.value,'=','::','std','::','primitive','::',primitive,';'].every((word,at)=>words[at]===word))throw new TypeError('Complete original monomorphic native scalar alias required');run=createRulePipeline([{phase:'language',rules:[{id:'owned-scalar-type:transport-complete-native-alias-library',apply:()=>{const concrete=transportConcreteSyntaxSource(selected.source,{language:'Rust'});if(concrete.source!==FIXED_TYPE_CORE_NATIVE_SOURCE||transportConcreteSyntaxMeta(concrete.meta).source!==selected.source)throw new TypeError('Complete original owned fixed type library differs');return concrete;}}]},{phase:'repository',rules:[{id:'router-core:bind-only-genuine-qualified-fixed-type-interface',apply:()=>{const type=primitive==='bool'?{...BOOL}:primitive==='f64'?{kind:'float'}:rustFixedType(primitive);if(!type)throw new TypeError('Existing checked scalar domain required');return {...type,rustPrimitiveName:primitive,rustSourceType:true};}}]}]).run(null);const current=select();if(current.source!==selected.source||current.file!==selected.file||current.rootFile!==selected.rootFile||current.span[0]!==selected.span[0]||current.span[1]!==selected.span[1]||nativeModule.sourceView!==view||nativeModule.source!==selected.source||!originalSourceModuleItems(nativeModule).includes(owner)||matchesParserFacts(parser)!==before||parser.cursor!==cursor||cursor.tokens!==tokens||cursor.index!==start||actual.some((token,at)=>tokens[start+at]!==token))throw new TypeError('Actual original owned fixed type source/parser association changed');}catch(error){if(!(error instanceof TypeError))throw error;throw unsupported('owned primitive type',error.message,location);}
 cursor.index=start+5;return run.value;
}
