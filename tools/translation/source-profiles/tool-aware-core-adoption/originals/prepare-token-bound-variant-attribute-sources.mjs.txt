const ROOT=fileURLToPath(new URL('../../',import.meta.url));
export function prepareTokenBoundVariantAttributeSources({root=ROOT}={}){
 return withNativeSubstitutionOperation(()=>prepareNativeSubstitutionCopy({root,archive:'tools/translation/source-profiles/token-bound-variant-attributes/complete-originals-and-rules.json.gz',pin:'37a78d31f2e5242a3d6462680bd3d5c4c4c0cab266b840a0bd2054d893f50f05',id:'router:token-bound-variant-attributes-complete-source-substitutions-v1'}));
}

 const args=process.argv.slice(2);if(args.length!==1||!['--emit','--write','--check'].includes(args[0]))throw new TypeError('Use --emit|--write|--check');const plan=prepareTokenBoundVariantAttributeSources();if(args[0]==='--write')writeNativeSubstitutionCopy(plan,{root:ROOT});if(args[0]==='--check'&&[...plan.files,...plan.descendantFiles].some(row=>row.current!==row.currentAfter))throw new TypeError('Token bound variant attribute source copy differs');console.log(JSON.stringify(args[0]==='--emit'?plan:{schema:plan.schema,files:plan.files.map(row=>row.path),receivingExecuted:false}));
