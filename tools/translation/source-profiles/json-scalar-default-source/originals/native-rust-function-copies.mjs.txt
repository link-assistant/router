import {readFileSync} from 'node:fs';
import {gunzipSync} from 'node:zlib';
import {createHash} from 'node:crypto';
import {resolve,relative,isAbsolute} from 'node:path';
import {fileURLToPath} from 'node:url';
import {sourceItems} from './rust-structure.mjs';

import {transportConcreteSyntaxSource,transportConcreteSyntaxMeta,applyConcreteSyntaxSubstitutions,serializeConcreteSyntaxMeta} from './concrete-syntax-meta.mjs';
import {transportConcreteSourceEquations} from './concrete-source-equations.mjs';
const ROOT=fileURLToPath(new URL('../../',import.meta.url));
const hash=value=>createHash('sha256').update(value).digest('hex');
const need=(value,message)=>{if(!value)throw new TypeError('Native Rust function source copies: '+message);};
function local(root,path){need(typeof path==='string'&&path.length>0&&!isAbsolute(path)&&!/[\\\0:]/u.test(path)&&path.split('/').every(part=>part&&part!=='.'&&part!=='..'),'repository-relative complete source path');const absolute=resolve(root,path),inside=relative(root,absolute);need(inside&&!inside.startsWith('../')&&!isAbsolute(inside),'source remains inside root');return absolute;}
export function prepareNativeRustFunctionCopies({root=ROOT,archive,pin,id}={}){
 root=resolve(root);need(typeof pin==='string'&&/^[0-9a-f]{64}$/u.test(pin)&&typeof id==='string'&&id.length>0&&id.length<=160,'explicit source profile identity');const archivePath=local(root,archive),packed=readFileSync(archivePath);need(packed.length<=2097152&&hash(packed)===pin,'complete pinned source/template profile');const profile=JSON.parse(gunzipSync(packed,{maxOutputLength:2097152}).toString('utf8'));
 need(profile.schema==='native-rust-function-copy-source-v1'&&profile.id===id&&typeof profile.path==='string'&&profile.path.endsWith('.rs')&&Array.isArray(profile.clones)&&profile.clones.length>0&&profile.clones.length<=128,'bounded complete native function source family');const target=local(root,profile.path),originalPath=local(root,profile.originalPath),before=profile.before;need(typeof before==='string'&&Buffer.byteLength(before)===profile.bytes&&hash(before)===profile.sha256&&readFileSync(originalPath,'utf8')===before,'complete immutable native original differs');need(Number.isSafeInteger(profile.originalFunctionCount)&&profile.originalFunctionCount>=1&&profile.originalFunctionCount<=2048&&profile.afterFunctionCount===profile.originalFunctionCount+profile.clones.length,'complete finite function census');
 const originals=sourceItems(before).filter(item=>item.term==='fn');need(originals.length===profile.originalFunctionCount,'complete original function census differs');const copies=[],edits=[];let offset=before.length+1,retained=Buffer.byteLength(before);
 for(const row of profile.clones){
  need(typeof row.name==='string'&&/^[\p{ID_Start}_][\p{ID_Continue}_]*$/u.test(row.name)&&typeof row.originalName==='string'&&typeof row.original==='string'&&typeof row.language==='string'&&Array.isArray(row.edits)&&row.edits.length<=64,'complete selected function and bounded edit program');const found=originals.filter(item=>item.name===row.originalName);need(found.length===1&&before.slice(found[0].start,found[0].end)===row.original,'whole original selected function differs');
  for(const edit of row.edits){need(Array.isArray(edit.span)&&edit.span.length===2&&Number.isSafeInteger(edit.span[0])&&Number.isSafeInteger(edit.span[1])&&edit.span[0]>=0&&edit.span[1]>=edit.span[0]&&edit.span[1]<=row.original.length&&typeof edit.expected==='string'&&typeof edit.replacement==='string'&&row.original.slice(...edit.span)===edit.expected,'exact complete native function edit');edits.push({...edit,span:edit.span.map(value=>value+offset)});need(edits.length<=512,'complete source edit count bound');}
  retained+=Buffer.byteLength(row.original)+Buffer.byteLength(row.language);need(retained<=2097152,'complete retained native source bound');copies.push(row.original);offset+=row.original.length+2;
 }
 const concrete=transportConcreteSyntaxSource(before,{language:'Rust'}),suffix='\n'+copies.join('\n\n')+'\n',languageRun=applyConcreteSyntaxSubstitutions(concrete.document,{languageRules:[{id:'rust:copy-complete-original-native-functions',edits:[{span:[before.length,before.length],expected:'',replacement:suffix}]},{id:'rust:substitute-exact-selected-native-function-source',edits}],repositoryRules:[{id:'router:retain-language-function-source-proposal',edits:[]}]}),language=transportConcreteSyntaxMeta(serializeConcreteSyntaxMeta(languageRun.value)),functions=sourceItems(language.source).filter(item=>item.term==='fn'&&item.start>=before.length);need(functions.length===profile.clones.length,'complete appended function census');
 const names=functions.map((item,index)=>{const row=profile.clones[index],header=originalDeclarationHeader(item,[]);need(item.name===row.originalName&&language.source.slice(item.start,item.end)===row.language&&header?.nameSpan,'complete transformed function and original name token');const span=[header.nameSpan.start,header.nameSpan.end];need(language.source.slice(...span)===row.originalName,'exact original function name source');return {span,expected:row.originalName,replacement:row.name};});
 const repositoryRun=applyConcreteSyntaxSubstitutions(language.document,{languageRules:[{id:'rust:retain-complete-transformed-native-functions',edits:[]}],repositoryRules:[{id:'router:publish-exact-native-compatibility-function-names',edits:names}]}),output=transportConcreteSyntaxMeta(serializeConcreteSyntaxMeta(repositoryRun.value)),after=output.source,afterFunctions=sourceItems(after).filter(item=>item.term==='fn');need(typeof profile.after==='string'&&after===profile.after&&afterFunctions.length===profile.afterFunctionCount&&new Set(afterFunctions.map(item=>item.name)).size===afterFunctions.length&&before.split('\n').length<1000&&after.split('\n').length<1000,'complete native output and file convention');
 const stages=transportConcreteSourceEquations([{before,after:language.source,language:'Rust'},{before:language.source,after,language:'Rust'}]).map((stage,index)=>({...stage,phase:index?'repository':'language'}));need(stages[0].inverse===before&&stages[1].inverse===language.source,'full native source inverses');const current=readFileSync(target,'utf8');need(current===before||current===after,'unknown current native source');need(readFileSync(archivePath).equals(packed)&&readFileSync(originalPath,'utf8')===before&&readFileSync(target,'utf8')===current,'complete original/native source changed during operation');
 return {schema:'native-substitution-source-copy-plan-v1',id,files:[{path:profile.path,originalPath:profile.originalPath,before,language:language.source,after,current,currentAfter:after,stages,inverse:before,templateInputs:profile.clones,sourceAuthority:false,semanticAuthority:false,nativeRustGrammarChecked:false,runtimeAuthority:false}],sourceAuthority:false,semanticAuthority:false,receivingExecuted:false,rustExecuted:false};
}
