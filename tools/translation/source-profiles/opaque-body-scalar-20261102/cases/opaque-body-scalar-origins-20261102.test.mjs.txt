// Predicate/source DATA units only. Synthetic proof records issue no authority.
import test from 'node:test';
import assert from 'node:assert/strict';
import {collectOpaqueBodyScalarOriginData} from '../opaque-body-scalar-origins-20261102.mjs';
function fixture(op,native,type={kind:'fixed',rustPrimitiveName:'u64',bits:64,signed:false}){const source='pub fn permit(a:u64,b:u64)->bool{a '+native+' b}',left=source.indexOf('{a ')+1,right=source.lastIndexOf(' b}')+1,proof={schema:'original-checked-executable-source-v1',source:'rust/src/permit.rs',qualifiedName:'permit.permit',span:{start:0,end:source.length},checkedSpan:{start:0,end:source.length},declaration:{body:{k:'binary',op,type:{kind:'bool'},left:{k:'var',type,span:{start:left,end:left+1}},right:{k:'var',type,span:{start:right,end:right+1}}}}},row={source:proof.source,qualifiedName:proof.qualifiedName,checkedSource:()=>proof,targetBinding:()=>({schema:'original-checked-project-target-binding-v1',source:proof.source,qualifiedName:proof.qualifiedName,javascriptName:'permit_permit'})};return {source,proof,fresh:{executableDeclarations:[row],meta:{executable:[{source:proof.source,name:proof.qualifiedName,term:'fn'}]}}};}
const operations=[['eq','==','===','equality'],['ne','!=','!==','equality'],['lt','<','<','comparison'],['le','<=','<=','comparison'],['gt','>','>','comparison'],['ge','>=','>=','comparison'],['and','&&','&&','logic'],['or','||','||','logic']];
test('all eight native predicate source tokens retain independent expected target spellings and full origins',()=>{for(const [op,native,target,kind]of operations){const value=fixture(op,native,kind==='logic'?{kind:'bool'}:undefined),rows=collectOpaqueBodyScalarOriginData(value.fresh,()=>value.source);assert.equal(rows.length,1,op);assert.equal(rows[0].kind,kind);assert.equal(rows[0].value,target);assert.equal(rows[0].nativeToken,native);assert.equal(rows[0].before,native);assert.equal(value.source.slice(rows[0].span.start,rows[0].span.end),native);assert.equal(rows[0].sourceAuthority,false);assert.equal(rows[0].semanticAuthority,false);}});
test('wrong original operator, unsupported scalar comparisons and unresolved types do not become origins',()=>{for(const value of [fixture('eq','!='),fixture('lt','<',{kind:'string'}),fixture('and','&&'),fixture('eq','==',{kind:'fixed',rustPrimitiveName:'custom',bits:64,signed:false}),fixture('eq','==',{kind:'bool',rustBorrowingUnresolved:true})])assert.deepEqual(collectOpaqueBodyScalarOriginData(value.fresh,()=>value.source),[]);});
test('comments between original operands cannot hide an extra operator or alter the source interval',()=>{const value=fixture('le','<=');value.source=value.source.replace(' <= ',' /* kept */ <= ');const right=value.source.lastIndexOf(' b}')+1;value.proof.span.end=value.source.length;value.proof.checkedSpan.end=value.source.length;value.proof.declaration.body.right.span={start:right,end:right+1};const rows=collectOpaqueBodyScalarOriginData(value.fresh,()=>value.source);assert.equal(rows.length,1);assert.equal(rows[0].before,'<=');assert.equal(rows[0].value,'<=');value.source=value.source.replace('<=','< +');value.proof.declaration.body.right.span.start++;value.proof.declaration.body.right.span.end++;value.proof.span.end++;value.proof.checkedSpan.end++;assert.deepEqual(collectOpaqueBodyScalarOriginData(value.fresh,()=>value.source),[]);});
test('fresh source readers preserve every primary value and no result survives another invocation',()=>{for(const primary of [undefined,null,false,0,'',Error('source')]){const value=fixture('eq','==');let caught=false;try{collectOpaqueBodyScalarOriginData(value.fresh,()=>{throw primary;});}catch(error){caught=true;assert.strictEqual(error,primary);}assert.equal(caught,true);}const one=fixture('eq','=='),two=fixture('ne','!=');assert.equal(collectOpaqueBodyScalarOriginData(one.fresh,()=>one.source)[0].value,'===');assert.equal(collectOpaqueBodyScalarOriginData(two.fresh,()=>two.source)[0].value,'!==');});
