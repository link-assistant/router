// Fresh version-control source DATA only. No stored owner/verdict cache.
import {execFileSync} from 'node:child_process';
const FILE=8*1024*1024,TOTAL=64*1024*1024,COUNT=4096;
const need=(value,message)=>{if(!value)throw new TypeError('Source unit Git baseline: '+message);};
export function captureSourceUnitGitBaseline(root,revision,{prefix,paths}){
 need(/^(?:[a-f0-9]{40}|[a-f0-9]{64})$/u.test(revision),'complete actual revision required');const wanted=new Set(paths),env={...Object.fromEntries(Object.entries(process.env).filter(([name])=>!name.startsWith('GIT_'))),GIT_CONFIG_NOSYSTEM:'1',GIT_CONFIG_GLOBAL:process.platform==='win32'?'NUL':'/dev/null',GIT_CONFIG_SYSTEM:process.platform==='win32'?'NUL':'/dev/null',GIT_LITERAL_PATHSPECS:'1',GIT_NO_LAZY_FETCH:'1'},git=args=>[process.platform==='win32'?'git':'/usr/bin/git',['-C',root,...args]],request=[...new Set([prefix,...paths.filter(path=>!path.startsWith(prefix+'/'))])];
 for(const path of request)need(typeof path==='string'&&path.length>0&&path.length<=4096&&!path.startsWith('/')&&!/[\\:\0\r\n]/u.test(path)&&path.split('/').every(part=>part&&part!=='.'&&part!=='..'),'canonical actual Git source paths required');
 const [program,args]=git(['ls-tree','-r','-l','-z',revision,'--',...request]),listing=execFileSync(program,args,{env,encoding:'utf8',timeout:30000,maxBuffer:8*1024*1024}),rows=[],objects=new Map();let total=0;
 for(const record of listing.split('\0').filter(Boolean)){const match=/^(\d{6}) (\S+) ([a-f0-9]+)\s+(\d+|-)\t([\s\S]+)$/u.exec(record);need(match,'complete native Git tree record required');const [,mode,type,oid,sizeText,path]=match;if(!wanted.has(path)&&!(path.startsWith(prefix+'/')&&path.endsWith('.rs')))continue;need(type==='blob'&&['100644','100755'].includes(mode)&&new RegExp('^[a-f0-9]{'+revision.length+'}$','u').test(oid)&&/^\d+$/u.test(sizeText),'committed source must be an ordinary full blob');const size=Number(sizeText);need(Number.isSafeInteger(size)&&size<=FILE,'committed original source file limit');need(path.length<=4096&&!/[\\:\0\r\n]/u.test(path)&&path.split('/').every(part=>part&&part!=='.'&&part!=='..'),'canonical committed source path required');rows.push({path,mode,oid,size});need(rows.length<=COUNT,'complete committed source count limit');if(!objects.has(oid)){objects.set(oid,{size});total+=size;need(total<=TOTAL,'complete original Git source byte limit');}}
 if(!rows.length)return new Map();const ids=[...objects.keys()],[batchProgram,batchArgs]=git(['cat-file','--batch']),bytes=execFileSync(batchProgram,batchArgs,{env,input:ids.join('\n')+'\n',timeout:30000,maxBuffer:TOTAL+1024*1024});let at=0;
 for(const oid of ids){const end=bytes.indexOf(10,at);need(end>=at&&end-at<=256,'complete Git batch header required');const header=bytes.subarray(at,end).toString('ascii'),match=/^([a-f0-9]+) blob (\d+)$/u.exec(header);need(match&&match[1]===oid&&Number(match[2])===objects.get(oid).size,'original Git batch object identity/size differs');const start=end+1,stop=start+objects.get(oid).size;need(stop<bytes.length&&bytes[stop]===10,'complete original Git source bytes required');const raw=bytes.subarray(start,stop),source=raw.toString('utf8');need(Buffer.from(source,'utf8').equals(raw),'lossless original committed UTF8 required');objects.get(oid).source=source;at=stop+1;}need(at===bytes.length,'unexpected original Git batch suffix');return new Map(rows.map(row=>[row.path,{source:objects.get(row.oid).source,mode:parseInt(row.mode,8)&0o777}]));
}
export function chooseSourceUnitAuthority(unit,current,baseline,nativePaths){
 const changed=[];if(nativePaths.some(path=>current.get(path)!==(baseline.get(path)?.source??null)))changed.push('Rust');for(const language of ['JavaScript','TypeScript']){const path=unit[language==='JavaScript'?'javascript':'typescript'];if(current.get(path)!==(baseline.get(path)?.source??null))changed.push(language);}if(changed.length>1)throw new TypeError('Concurrent source edits require reconciliation: '+changed.join(', '));const selected=unit.authority;
 if(selected!=='auto'){if(changed.length&&changed[0]!==selected)throw new TypeError('Edited non-authoritative source would be overwritten');if(selected!=='Rust'&&current.get(unit[selected==='JavaScript'?'javascript':'typescript'])===null)throw new TypeError('Actual authoritative source is missing');return selected;}
 if(changed.length===1){if(changed[0]!=='Rust'&&current.get(unit[changed[0]==='JavaScript'?'javascript':'typescript'])===null)throw new TypeError('Source deletion is not an implicit translation request');return changed[0];}const available=['Rust',...['JavaScript','TypeScript'].filter(language=>current.get(unit[language==='JavaScript'?'javascript':'typescript'])!==null)];if(available.length!==1)throw new TypeError('Unchanged multi-language unit requires explicit authority');return available[0];
}
