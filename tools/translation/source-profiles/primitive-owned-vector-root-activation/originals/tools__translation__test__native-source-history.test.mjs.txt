// Independent native history construction/source freshness. SOURCE ONLY; UNRUN.
import test from 'node:test';import assert from 'node:assert/strict';
import {readFileSync,writeFileSync,mkdirSync,mkdtempSync,rmSync} from 'node:fs';import {join,dirname} from 'node:path';import {tmpdir} from 'node:os';import {gunzipSync,gzipSync} from 'node:zlib';import {createHash} from 'node:crypto';
import {prepareNativeSourceHistory,currentNativeSourceProvenance,assertNativeSourceProvenanceCurrent} from '../native-source-history.mjs';
import {prepareNativeCatalogAlignmentSources} from '../prepare-native-catalog-alignment-sources.mjs';
const limits={timeout:60000},archive='tools/translation/source-profiles/primitive-sequence-root-activation/complete-native-history.json.gz',root=new URL('../../../',import.meta.url);
function fixture(t){const directory=mkdtempSync(join(tmpdir(),'router-native-history-')),packed=readFileSync(new URL(archive,root)),rows=JSON.parse(gunzipSync(packed).toString('utf8')).files;t.after(()=>rmSync(directory,{recursive:true,force:true}));mkdirSync(dirname(join(directory,archive)),{recursive:true});writeFileSync(join(directory,archive),packed);for(const row of rows){mkdirSync(dirname(join(directory,row.path)),{recursive:true});mkdirSync(dirname(join(directory,row.originalPath)),{recursive:true});writeFileSync(join(directory,row.path),row.after);writeFileSync(join(directory,row.originalPath),row.before);}return {directory,rows};}
test('complete native paths retain language identity repository edits exact originals current source and full two-stage CST inverses',limits,t=>{
 const {directory,rows}=fixture(t),plan=prepareNativeSourceHistory({root:directory});assert.equal(plan.files.length,17);for(const row of plan.files){assert.equal(row.current,row.after);assert.equal(row.inverse,row.before);assert.deepEqual(row.stages.map(stage=>stage.phase),row.path==='rust/src/translation_json_core.rs'?['language','repository','language','repository','language','repository']:['rust/src/lib.rs','rust/src/operation_context.rs','rust/src/with_command.rs'].includes(row.path)?['language','repository','language','repository']:['language','repository']);assert.equal(row.sourceAuthority,false);assert.equal(row.semanticAuthority,false);assert.equal(row.namespaceAuthority,false);}assert.equal(plan.sourceCache,false);assert.equal(plan.rustExecuted,false);assert.equal(plan.receivingExecuted,false);for(const row of rows){const joined=currentNativeSourceProvenance(directory,{...row,source:row.before});assert.equal(joined.originalSource,row.before);assert.equal(joined.source,readFileSync(join(directory,row.path),'utf8'));assert.equal(joined.nativeSourceHistory.inverse,row.before);assertNativeSourceProvenanceCurrent(directory,joined);assert.equal(readFileSync(join(directory,row.path),'utf8'),joined.source);}
});
test('unchanged native inputs stay unchanged and old-to-current history cannot accept unknown bytes or changed original inputs',limits,t=>{
 const {directory,rows}=fixture(t),row=rows[0];writeFileSync(join(directory,row.path),row.before);const same=currentNativeSourceProvenance(directory,{...row,source:row.before});assert.equal(same.source,row.before);assert.equal(same.nativeSourceHistory,null);writeFileSync(join(directory,row.path),row.after);const current=currentNativeSourceProvenance(directory,{...row,source:row.before});writeFileSync(join(directory,row.path),row.after+'\n// unknown successor');assert.throws(()=>assertNativeSourceProvenanceCurrent(directory,current),/unknown|equation|changed/);writeFileSync(join(directory,row.path),row.after);writeFileSync(join(directory,row.originalPath),row.before+'\n');assert.throws(()=>currentNativeSourceProvenance(directory,{...row,source:row.before}),/original/);
});
test('complete history archive tamper and wrong language cannot be laundered into changed native provenance',limits,t=>{
 const {directory,rows}=fixture(t),row=rows[0];assert.throws(()=>currentNativeSourceProvenance(directory,{...row,source:row.before,language:'JavaScript'}),/language/);const path=join(directory,archive),bytes=readFileSync(path);writeFileSync(path,Buffer.concat([bytes,Buffer.from([0])]));assert.throws(()=>currentNativeSourceProvenance(directory,{...row,source:row.before}),/pinned/);
});
test('seven compiler equations retain original read guards while explicit native successors carry both complete Rust source histories',limits,()=>{
 const plan=prepareNativeCatalogAlignmentSources();assert.equal(plan.files.length,7);for(const row of plan.files){assert.equal(row.current,row.currentAfter);assert.equal(row.inverse,row.before);assert.deepEqual(row.stages.map(stage=>stage.phase),['language','repository']);}assert.equal(plan.nativeInputs.files.length,17);assert.equal(plan.nativeInputs.sourceCache,false);assert.equal(plan.rustExecuted,false);assert.equal(plan.receivingExecuted,false);
});

test('receiving DATA histories compose multiple complete source versions while branches remain explicit refusals',limits,t=>{
 const directory=mkdtempSync(join(tmpdir(),'router-native-history-chain-'));t.after(()=>rmSync(directory,{recursive:true,force:true}));const before='pub fn selected()->u64{1}\n',middle='pub fn selected()->u64{2}\n',after='pub fn selected()->u64{3}\n',path='rust/src/selected.rs',archive='profile/native-history.json.gz',hash=value=>createHash('sha256').update(value).digest('hex');mkdirSync(dirname(join(directory,path)),{recursive:true});mkdirSync(join(directory,'profile'));writeFileSync(join(directory,path),after);
 const row=(source,next,index)=>{const originalPath='profile/original-'+index+'.txt';writeFileSync(join(directory,originalPath),source);const expected=index===0?'1':'2',start=source.indexOf(expected);return {path,language:'Rust',originalPath,before:source,after:next,beforeSha256:hash(source),afterSha256:hash(next),languageEdits:[],repositoryEdits:[{span:[start,start+1],expected,replacement:index===0?'2':'3'}],ruleId:'receiving:step-'+index};},rows=[row(before,middle,0),row(middle,after,1)],packed=gzipSync(JSON.stringify({schema:'native-complete-source-history-v1',files:rows}));writeFileSync(join(directory,archive),packed);const plan=prepareNativeSourceHistory({root:directory,archive,pin:hash(packed)});assert.equal(plan.files.length,1);assert.equal(plan.files[0].historyChain.length,2);assert.equal(plan.files[0].before,before);assert.equal(plan.files[0].after,after);assert.equal(plan.files[0].inverse,before);assert.deepEqual(plan.files[0].stages.map(stage=>stage.phase),['language','repository','language','repository']);assert.equal(plan.sourceAuthority,false);assert.equal(plan.semanticAuthority,false);assert.equal(plan.sourceCache,false);
 const branch={...rows[1],before,beforeSha256:hash(before),repositoryEdits:[{span:[before.indexOf('1'),before.indexOf('1')+1],expected:'1',replacement:'3'}]};writeFileSync(join(directory,branch.originalPath),before);const changed=gzipSync(JSON.stringify({schema:'native-complete-source-history-v1',files:[rows[0],branch]}));writeFileSync(join(directory,archive),changed);assert.throws(()=>prepareNativeSourceHistory({root:directory,archive,pin:hash(changed)}),/head|branch|duplicate/);assert.equal(readFileSync(join(directory,path),'utf8'),after);
});
