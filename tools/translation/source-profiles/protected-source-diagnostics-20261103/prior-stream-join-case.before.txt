const root=fileURLToPath(new URL('../../../',import.meta.url)),base='tools/translation/source-profiles/protected-stream-composed-current-20261103/',read=path=>readFileSync(new URL('../../../'+path,import.meta.url),'utf8'),parents=[["stream-history","tools/translation/protected-native-stream-fds-20261103-history.mjs"]];
const input=parents.map(([role,path])=>({path,source:read(base+role+'.before.txt')}));

test('one complete named stream-history parent retain full ordered source equations and complete inverses',{timeout:60000},()=>{const proposal=proposeProtectedStreamComposedCurrentSources(input,{root});assert.equal(proposal.files.length,1);assert.equal(proposal.sourceInstalled,false);assert.equal(proposal.verification,'UNRUN');for(const [index,[role,path]]of parents.entries()){const row=proposal.files[index];assert.equal(row.path,path);assert.equal(row.before,input[index].source);assert.equal(row.after,read(base+role+'.after.txt'));assert.equal(row.inverse,row.before);assert.deepEqual(row.trace.map(stage=>stage.phase),['language','repository']);for(const stage of row.stages){assert.equal(transportConcreteSyntaxMeta(stage.beforeMeta).source,stage.before);assert.equal(transportConcreteSyntaxMeta(stage.afterMeta).source,stage.after);assert.equal(stage.inverse,stage.before);}const current=knownProtectedStreamComposedCurrentSuccessor({root,path,before:row.before});assert.equal(current.currentAfter,read(path));assert.equal(knownProtectedStreamComposedCurrentSuccessor({root,path,before:row.before+'\n'}),null);}assert.equal(knownProtectedStreamComposedCurrentSuccessor({root,path:'unassigned.mjs',before:''}),null);assert.throws(()=>proposeProtectedStreamComposedCurrentSources(input.slice(1),{root}),/all one/u);});
